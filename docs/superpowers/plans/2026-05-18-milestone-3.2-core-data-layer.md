# Milestone 3.2 — Core Data Layer + SQLCipher Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the production-grade core data layer (14 services + `HabitConfigHistory` interceptor + soft-delete/restore/purge + SQLCipher with DPAPI/passphrase providers and key transition + p95<300ms home-tree read budget).

**Architecture:** Service-per-aggregate in `Recordum.Data/Services/` (scoped DI, `IPluginContext` capture), `HabitConfigHistoryInterceptor` registered before `TimestampInterceptor`, soft-delete via shared timestamp + recursive CTE, encryption-at-rest via `SQLitePCLRaw.bundle_e_sqlcipher` driven by `IDbKeyProvider` (DPAPI default, Argon2id passphrase opt-in), perf budget asserted as xUnit `[Fact]` on deterministic 200-habit/50k-completion seed.

**Tech Stack:** C# / .NET 10, EF Core 9 + SQLite, `SQLitePCLRaw.bundle_e_sqlcipher`, `Konscious.Security.Cryptography.Argon2`, `Microsoft.AspNetCore.DataProtection`/`System.Security.Cryptography.ProtectedData`, xUnit + `Microsoft.Extensions.DependencyInjection`.

**Spec:** `docs/superpowers/specs/2026-05-18-milestone-3.2-core-data-layer-design.md` is authoritative when it conflicts with this plan.

```csharp
// KeyMaterial.cs
using System.Security.Cryptography;
namespace Recordum.Core.Abstractions.Security;
public sealed class KeyMaterial : IDisposable
{
    public const int LengthBytes = 32;
    public byte[] RawKey { get; }
    public KeyMaterial(byte[] rawKey)
    {
        ArgumentNullException.ThrowIfNull(rawKey);
        if (rawKey.Length != LengthBytes)
            throw new ArgumentException($"Key must be {LengthBytes} bytes (got {rawKey.Length}).", nameof(rawKey));
        RawKey = rawKey;
    }
    public void Dispose() => CryptographicOperations.ZeroMemory(RawKey);
}
```

```csharp
    {
```

```csharp
// HabitConfigHistoryInterceptor.cs
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.ChangeTracking;
using Microsoft.EntityFrameworkCore.Diagnostics;
using Microsoft.Extensions.Logging;
using Recordum.Core.Abstractions;
using Recordum.Core.Models;

namespace Recordum.Data.Interceptors;

public sealed class HabitConfigHistoryInterceptor(
    IClock clock,
    ILogger<HabitConfigHistoryInterceptor> logger) : SaveChangesInterceptor
{
    public override InterceptionResult<int> SavingChanges(DbContextEventData e, InterceptionResult<int> r)
    { Apply(e.Context); return base.SavingChanges(e, r); }
    public override ValueTask<InterceptionResult<int>> SavingChangesAsync(DbContextEventData e, InterceptionResult<int> r, CancellationToken ct = default)
    { Apply(e.Context); return base.SavingChangesAsync(e, r, ct); }

    private void Apply(DbContext? db)
    {
        if (db is null) return;
        var entries = db.ChangeTracker.Entries<HabitConfig>()
            .Where(en => en.State == EntityState.Modified || en.State == EntityState.Deleted)
            .ToList();
        if (entries.Count == 0) return;

        var nowUtc = clock.UtcNow.UtcDateTime;
        foreach (var entry in entries)
        {
            if (!HasVersionedChange(entry)) continue;
            try
            {
                var snap = BuildSnapshot(entry, nowUtc);
                db.Set<HabitConfigHistory>().Add(snap);
            }
            catch (InvalidOperationException ex)
            {
                logger.LogWarning(ex, "Skipping history snapshot for HabitConfig {Id}: original values unavailable", entry.Entity.Id);
            }
        }
    }

    private static bool HasVersionedChange(EntityEntry<HabitConfig> entry)
    {
        if (entry.State == EntityState.Deleted) return true;
        foreach (var name in HabitConfigHistoryColumns.Versioned)
        {
            var p = entry.Property(name);
            if (!Equals(p.OriginalValue, p.CurrentValue)) return true;
        }
        return false;
    }

    private static HabitConfigHistory BuildSnapshot(EntityEntry<HabitConfig> entry, DateTime nowUtc)
    {
        var original = entry.OriginalValues;
        var snap = new HabitConfigHistory
        {
            Id = Guid.NewGuid(),
            HabitNodeId = entry.Entity.HabitNodeId,
            ValidFrom = (DateTime)original[nameof(HabitConfig.UpdatedAt)]!,
            ValidUntil = nowUtc,
        };
        foreach (var name in HabitConfigHistoryColumns.Versioned)
        {
            var src = original[name];
            typeof(HabitConfigHistory).GetProperty(name)!.SetValue(snap, src);
        }
        return snap;
    }
}
```

```csharp
    public async ValueTask<Guid> CreateAsync(HabitNode draft, CancellationToken ct = default)
    {
        ArgumentNullException.ThrowIfNull(draft);
        if (draft.Id == Guid.Empty) draft.Id = Guid.NewGuid();
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        db.HabitNodes.Add(draft);
        await db.SaveChangesAsync(ct);
        logger.LogDebug("HabitNode {Id} created (plugin={PluginId})", draft.Id, pluginContext.PluginId);
        return draft.Id;
    }

    public async ValueTask UpdateAsync(Guid id, Action<HabitNode> mutate, CancellationToken ct = default)
    {
        ArgumentNullException.ThrowIfNull(mutate);
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        var node = await db.HabitNodes.FirstOrDefaultAsync(n => n.Id == id, ct)
            ?? throw new HabitNodeNotFoundException(id);
        if (node.DeletedAt is not null) throw new HabitNodeAlreadySoftDeletedException(id);
        mutate(node);
        await db.SaveChangesAsync(ct);
        logger.LogDebug("HabitNode {Id} updated", id);
    }

    public async ValueTask MoveAsync(Guid id, Guid? newParentId, CancellationToken ct = default)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        var node = await db.HabitNodes.FirstOrDefaultAsync(n => n.Id == id, ct)
            ?? throw new HabitNodeNotFoundException(id);
        if (node.DeletedAt is not null) throw new HabitNodeAlreadySoftDeletedException(id);
        if (newParentId is { } np)
        {
            // Cycle guard: descendants of `id` cannot become parents of `id`.
            var descendantIds = await DescendantIdsAsync(db, id, ct);
            if (np == id || descendantIds.Contains(np))
                throw new CyclicHabitNodeMoveException(id, np);
            var parentExists = await db.HabitNodes.AnyAsync(n => n.Id == np, ct);
            if (!parentExists) throw new HabitNodeNotFoundException(np);
        }
        node.ParentId = newParentId;
        await db.SaveChangesAsync(ct);
        logger.LogInformation("HabitNode {Id} moved to {NewParent}", id, newParentId);
    }

    public async ValueTask<HabitNode?> GetAsync(Guid id, CancellationToken ct = default)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        return await db.HabitNodes.AsNoTracking().FirstOrDefaultAsync(n => n.Id == id, ct);
    }

    public ValueTask SoftDeleteAsync(Guid id, CancellationToken ct = default) => throw new NotImplementedException("Task 10");
    public ValueTask RestoreAsync(Guid id, CancellationToken ct = default) => throw new NotImplementedException("Task 10");

    internal static async Task<HashSet<Guid>> DescendantIdsAsync(RecordumDbContext db, Guid rootId, CancellationToken ct)
    {
        // Recursive CTE — fast on deep hierarchies.
        var conn = db.Database.GetDbConnection();
        if (conn.State != System.Data.ConnectionState.Open) await conn.OpenAsync(ct);
        await using var cmd = conn.CreateCommand();
        cmd.CommandText = """
            WITH RECURSIVE d(id) AS (
                SELECT id FROM habit_node WHERE id = $root
                UNION ALL
                SELECT n.id FROM habit_node n JOIN d ON n.parent_id = d.id
            )
            SELECT id FROM d WHERE id <> $root;
            """;
        var p = cmd.CreateParameter(); p.ParameterName = "$root"; p.Value = rootId.ToString(); cmd.Parameters.Add(p);
        var ids = new HashSet<Guid>();
        await using var reader = await cmd.ExecuteReaderAsync(ct);
        while (await reader.ReadAsync(ct))
            ids.Add(Guid.Parse(reader.GetString(0)));
        return ids;
    }
}
```

```csharp
public async ValueTask SoftDeleteAsync(Guid id, CancellationToken ct = default)
{
    await using var db = await dbFactory.CreateDbContextAsync(ct);
    var node = await db.HabitNodes.IgnoreQueryFilters()
        .FirstOrDefaultAsync(n => n.Id == id, ct)
        ?? throw new HabitNodeNotFoundException(id);
    if (node.DeletedAt is not null) return; // idempotent
    var deletedAt = clock.UtcNow.UtcDateTime;
    var descendants = await DescendantIdsAsync(db, id, ct);
    descendants.Add(id);
    await using var tx = await db.Database.BeginTransactionAsync(ct);
    await db.HabitNodes.IgnoreQueryFilters()
        .Where(n => descendants.Contains(n.Id) && n.DeletedAt == null)
        .ExecuteUpdateAsync(setters => setters.SetProperty(n => n.DeletedAt, deletedAt), ct);
    await tx.CommitAsync(ct);
    logger.LogInformation("HabitNode {Id} soft-deleted (subtree size {Count})", id, descendants.Count);
}
```

```csharp
using Microsoft.Extensions.Logging;
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Enums;
using Recordum.Core.Errors;
using Recordum.Core.Models;
namespace Recordum.Data.Services;

public sealed class CompletionService(
    IDbContextFactory<RecordumDbContext> dbFactory,
    IClock clock,
    ILogger<CompletionService> logger) : ICompletionService
{
    public async ValueTask<Guid> RecordAsync(Completion draft, IReadOnlyList<CompletionValue>? values = null, IReadOnlyList<CompletionSegment>? segments = null, CancellationToken ct = default)
    {
        if (draft.Id == Guid.Empty) draft.Id = Guid.NewGuid();
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        var habit = await db.HabitNodes.FirstOrDefaultAsync(n => n.Id == draft.HabitNodeId, ct)
            ?? throw new HabitNodeNotFoundException(draft.HabitNodeId);
        if (habit.HabitType is null)
            throw new InvalidCompletionForHabitTypeException(draft.HabitNodeId, default, "node is a group, not a habit");
        ValidatePerType(habit.HabitType.Value, draft, values, segments);
        db.Completions.Add(draft);
        if (values is not null) foreach (var v in values) { v.CompletionId = draft.Id; if (v.Id == Guid.Empty) v.Id = Guid.NewGuid(); db.CompletionValues.Add(v); }
        if (segments is not null) foreach (var s in segments) { s.CompletionId = draft.Id; if (s.Id == Guid.Empty) s.Id = Guid.NewGuid(); db.CompletionSegments.Add(s); }
        await db.SaveChangesAsync(ct);
        logger.LogInformation("Completion {Id} recorded for habit {HabitId}", draft.Id, draft.HabitNodeId);
        return draft.Id;
    }

    public async ValueTask UpdateAsync(Guid id, Action<Completion> mutate, CancellationToken ct = default)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        var c = await db.Completions.FirstOrDefaultAsync(x => x.Id == id, ct);
        if (c is null) return;
        mutate(c);
        await db.SaveChangesAsync(ct);
    }

    public async ValueTask DeleteAsync(Guid id, CancellationToken ct = default)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        await db.Completions.Where(x => x.Id == id).ExecuteDeleteAsync(ct);
    }

    public async ValueTask<Completion?> GetAsync(Guid id, CancellationToken ct = default)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        return await db.Completions.Include(c => c.Values).Include(c => c.Segments).AsNoTracking().FirstOrDefaultAsync(c => c.Id == id, ct);
    }

    private static void ValidatePerType(HabitType type, Completion c, IReadOnlyList<CompletionValue>? values, IReadOnlyList<CompletionSegment>? segments)
    {
        switch (type)
        {
            case HabitType.Action:
                // Completion stands on its own; no values required.
                break;
            case HabitType.Counter:
                if (values is null or { Count: 0 })
                    throw new InvalidCompletionForHabitTypeException(c.HabitNodeId, type, "Counter completion requires at least one CompletionValue");
                break;
            case HabitType.Resistance:
                if (values is null or { Count: 0 })
                    throw new InvalidCompletionForHabitTypeException(c.HabitNodeId, type, "Resistance completion requires at least one CompletionValue");
                break;
            case HabitType.Journal:
                if (string.IsNullOrWhiteSpace(c.Notes))
                    throw new InvalidCompletionForHabitTypeException(c.HabitNodeId, type, "Journal completion requires Notes");

                if (values is null or { Count: 0 })
```

```csharp
// PurgeService.cs
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Abstractions.Services.Dtos;

namespace Recordum.Data.Services;

public sealed class PurgeService(
    IDbContextFactory<RecordumDbContext> dbFactory,
    IClock clock,
    ILogger<PurgeService> logger) : IPurgeService
{
    public async ValueTask<PurgeReport> PurgeExpiredAsync(TimeSpan olderThan, CancellationToken ct = default)
    {
        var cutoff = clock.UtcNow.UtcDateTime - olderThan;
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        var rootIds = await db.HabitNodes.IgnoreQueryFilters()
            .Where(n => n.DeletedAt != null && n.DeletedAt < cutoff && n.ParentId == null)
            .Select(n => n.Id).ToListAsync(ct);
        var errors = new List<PurgeError>();
        int purged = 0, skipped = 0;

        foreach (var root in rootIds)
        {
            try
            {
                await using var tx = await db.Database.BeginTransactionAsync(ct);
                var subtree = await HabitNodeService.DescendantIdsAsync(db, root, ct);
                subtree.Add(root);
                await db.Attachments.Where(a => a.Note != null && subtree.Contains(a.Note.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.Notes.Where(n => subtree.Contains(n.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.CompletionValues.Where(v => subtree.Contains(v.Completion.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.CompletionSegments.Where(s => subtree.Contains(s.Completion.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.Completions.Where(c => subtree.Contains(c.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.Goals.Where(g => subtree.Contains(g.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.MetricDefinitions.Where(m => subtree.Contains(m.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.Notifications.Where(n => subtree.Contains(n.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.StreakPauses.Where(p => subtree.Contains(p.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.HabitNodeTags.Where(t => subtree.Contains(t.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.HabitConfigHistory.Where(h => subtree.Contains(h.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.HabitConfigs.Where(c => subtree.Contains(c.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.SegmentDefinitions.Where(s => subtree.Contains(s.HabitNodeId)).ExecuteDeleteAsync(ct);
                await db.HabitNodes.IgnoreQueryFilters().Where(n => subtree.Contains(n.Id)).ExecuteDeleteAsync(ct);
                await tx.CommitAsync(ct);
                purged++;
                logger.LogInformation("Purged subtree rooted at {Root} ({Size} nodes)", root, subtree.Count);
            }
            catch (Exception ex)
            {
                skipped++;
                errors.Add(new PurgeError(root, "delete", ex.Message));
                logger.LogError(ex, "Purge failed for {Root}", root);
            }
        }
        return new PurgeReport(purged, skipped, errors);
    }
}
```

```csharp
// DpapiKeyProvider.cs
using System.Security.Cryptography;
using Recordum.Core.Abstractions.Security;
using Recordum.Core.Errors;

namespace Recordum.Data.Security;

public sealed class DpapiKeyProvider(string keyFilePath) : IDbKeyProvider
{
    public string ProviderId => KeyProviderId.Dpapi;
    public async ValueTask<KeyMaterial> GetKeyAsync(CancellationToken ct = default)
    {
        if (!File.Exists(keyFilePath))
        {
            var fresh = new byte[KeyMaterial.LengthBytes];
            RandomNumberGenerator.Fill(fresh);
            var protectedBytes = ProtectedData.Protect(fresh, optionalEntropy: null, DataProtectionScope.CurrentUser);
            Directory.CreateDirectory(Path.GetDirectoryName(keyFilePath)!);
            await File.WriteAllBytesAsync(keyFilePath, protectedBytes, ct);
            return new KeyMaterial(fresh);
        }
        try
        {
            var blob = await File.ReadAllBytesAsync(keyFilePath, ct);
            var raw = ProtectedData.Unprotect(blob, optionalEntropy: null, DataProtectionScope.CurrentUser);
            return new KeyMaterial(raw);
        }
        catch (CryptographicException ex)
        {
            throw new KeyUnavailableException(ProviderId, $"cannot decrypt {keyFilePath}: {ex.Message}");
        }
    }
}
```

```csharp
// PassphraseKeyProvider.cs
using System.Text;
using Konscious.Security.Cryptography;
using Recordum.Core.Abstractions.Security;
using Recordum.Core.Errors;

namespace Recordum.Data.Security;

public sealed class PassphraseKeyProvider : IDbKeyProvider
{
    private const int MinPassphraseLength = 12;
    private readonly byte[] _passphraseBytes;
    private readonly string _saltPath;

    public PassphraseKeyProvider(string passphrase, string saltPath)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(passphrase);
        if (passphrase.Length < MinPassphraseLength)
            throw new ArgumentException($"Passphrase must be at least {MinPassphraseLength} characters.", nameof(passphrase));
        _passphraseBytes = Encoding.UTF8.GetBytes(passphrase);
        _saltPath = saltPath;
    }

    public string ProviderId => KeyProviderId.Passphrase;

    public async ValueTask<KeyMaterial> GetKeyAsync(CancellationToken ct = default)
    {
        if (!File.Exists(_saltPath))
            throw new KeyUnavailableException(ProviderId, $"salt file missing: {_saltPath}");
        var salt = await File.ReadAllBytesAsync(_saltPath, ct);
        if (salt.Length != 16) throw new KeyUnavailableException(ProviderId, $"unexpected salt length {salt.Length}");
        using var argon = new Argon2id(_passphraseBytes)
        {
            Salt = salt,
            DegreeOfParallelism = 2,
            Iterations = 4,
            MemorySize = 65_536,
        };
        var raw = await argon.GetBytesAsync(KeyMaterial.LengthBytes);
        return new KeyMaterial(raw);
    }
}
```

```csharp
// KeyTransitionService.cs
using System.Security.Cryptography;
using Microsoft.Extensions.Logging;
using Recordum.Core.Abstractions.Security;
using Recordum.Core.Errors;

namespace Recordum.Data.Security;

public sealed class KeyTransitionService(
    IDbStoragePaths paths,
    IDbKeyProvider currentProvider,
    ILogger<KeyTransitionService> logger) : IKeyTransitionService
{
    public async ValueTask TransitionToPassphraseAsync(string newPassphrase, CancellationToken ct = default)
    {
        if (string.IsNullOrWhiteSpace(newPassphrase) || newPassphrase.Length < 12)
            throw new ArgumentException("passphrase must be >= 12 chars", nameof(newPassphrase));
        var backup = paths.DatabasePath + $".bak.{DateTime.UtcNow.Ticks}";
        File.Copy(paths.DatabasePath, backup, overwrite: false);
        try
        {
            var salt = new byte[16]; RandomNumberGenerator.Fill(salt);
            using var currentKey = await currentProvider.GetKeyAsync(ct);
            // Derive new key via Argon2id without writing salt yet.
            using var argon = new Konscious.Security.Cryptography.Argon2id(System.Text.Encoding.UTF8.GetBytes(newPassphrase))
            { Salt = salt, DegreeOfParallelism = 2, Iterations = 4, MemorySize = 65_536 };
            var newRaw = await argon.GetBytesAsync(KeyMaterial.LengthBytes);
            using (var conn = SqlCipherConnectionFactory.Open(paths.DatabasePath, currentKey))
            using (var cmd = conn.CreateCommand())
            {
                cmd.CommandText = $"PRAGMA rekey = \"x'{Convert.ToHexString(newRaw)}'\";";
                await cmd.ExecuteNonQueryAsync(ct);
            }
            await File.WriteAllBytesAsync(paths.SaltFilePath, salt, ct);
            if (File.Exists(paths.KeyFilePath)) File.Delete(paths.KeyFilePath);
            File.Delete(backup);
            logger.LogInformation("Key transition DPAPI -> passphrase completed");
        }
        catch (Exception ex)
        {
            File.Copy(backup, paths.DatabasePath, overwrite: true);
            try { File.Delete(backup); } catch { }
            if (File.Exists(paths.SaltFilePath)) File.Delete(paths.SaltFilePath);
            throw new KeyTransitionFailedException(KeyProviderId.Dpapi, KeyProviderId.Passphrase, ex);
        }
    }

    public async ValueTask TransitionToDpapiAsync(string currentPassphrase, CancellationToken ct = default)
    {
        var backup = paths.DatabasePath + $".bak.{DateTime.UtcNow.Ticks}";
        File.Copy(paths.DatabasePath, backup, overwrite: false);
        try
        {
            using var currentKey = await new PassphraseKeyProvider(currentPassphrase, paths.SaltFilePath).GetKeyAsync(ct);
            var fresh = new byte[KeyMaterial.LengthBytes]; RandomNumberGenerator.Fill(fresh);
            using (var conn = SqlCipherConnectionFactory.Open(paths.DatabasePath, currentKey))
            using (var cmd = conn.CreateCommand())
            {
                cmd.CommandText = $"PRAGMA rekey = \"x'{Convert.ToHexString(fresh)}'\";";
                await cmd.ExecuteNonQueryAsync(ct);
            }
            var protectedBytes = ProtectedData.Protect(fresh, optionalEntropy: null, DataProtectionScope.CurrentUser);
            await File.WriteAllBytesAsync(paths.KeyFilePath, protectedBytes, ct);
            if (File.Exists(paths.SaltFilePath)) File.Delete(paths.SaltFilePath);
            File.Delete(backup);
            logger.LogInformation("Key transition passphrase -> DPAPI completed");
        }
        catch (Exception ex)
        {
            File.Copy(backup, paths.DatabasePath, overwrite: true);
            try { File.Delete(backup); } catch { }
            if (File.Exists(paths.KeyFilePath)) File.Delete(paths.KeyFilePath);
            throw new KeyTransitionFailedException(KeyProviderId.Passphrase, KeyProviderId.Dpapi, ex);
        }
    }
}
```

```csharp
// PerfDataSeeder.cs
using Microsoft.EntityFrameworkCore;
using Recordum.Core.Enums;
using Recordum.Core.Models;

namespace Recordum.Data.Tests.Performance;

public static class PerfDataSeeder
{
    public static async Task SeedAsync(IDbContextFactory<RecordumDbContext> factory, int seed = 42, CancellationToken ct = default)
    {
        var rnd = new Random(seed);
        await using var db = await factory.CreateDbContextAsync(ct);

        // 30 groups, max depth 3
        var groups = new List<HabitNode>();
        for (int i = 0; i < 30; i++)
        {
            Guid? parent = groups.Count > 0 && rnd.NextDouble() < 0.5 ? groups[rnd.Next(groups.Count)].Id : null;
            groups.Add(new HabitNode { Id = Guid.NewGuid(), NodeType = NodeType.Group, Name = $"g{i}", ParentId = parent });
        }
        db.HabitNodes.AddRange(groups);

        // 200 habits across the 5 types: 80/60/30/20/10
        var habitTypeMix = new (HabitType type, int count)[] {
            (HabitType.Action, 80), (HabitType.Counter, 60), (HabitType.Resistance, 30), (HabitType.Journal, 20), (HabitType.State, 10),
        };
        var habits = new List<HabitNode>();
        int hi = 0;
        foreach (var (type, count) in habitTypeMix)
            for (int j = 0; j < count; j++)
                habits.Add(new HabitNode {
                    Id = Guid.NewGuid(), NodeType = NodeType.Habit, HabitType = type,
                    Name = $"h{hi++}", ParentId = groups[rnd.Next(groups.Count)].Id,
                });
        db.HabitNodes.AddRange(habits);
        await db.SaveChangesAsync(ct);

        // 50k completions distributed over last 12 months.
        var horizonStart = new DateTime(2025, 5, 1, 0, 0, 0, DateTimeKind.Utc);
        var horizonMs = (long)(DateTime.UtcNow - horizonStart).TotalMilliseconds;
        var batch = new List<Completion>(5000);
        for (int i = 0; i < 50_000; i++)
        {
            var habit = habits[rnd.Next(habits.Count)];
            var offsetMs = (long)(rnd.NextDouble() * horizonMs);
            var when = horizonStart.AddMilliseconds(offsetMs);
            batch.Add(new Completion {
                Id = Guid.NewGuid(), HabitNodeId = habit.Id,
                CompletedAt = when, CompletedAtMs = ((DateTimeOffset)when).ToUnixTimeMilliseconds(),
                Notes = habit.HabitType == HabitType.Journal ? "x" : null,
            });
            if (batch.Count == 5000)
            {
                db.Completions.AddRange(batch);
                await db.SaveChangesAsync(ct);
                foreach (var c in batch) db.Entry(c).State = EntityState.Detached;
                batch.Clear();
            }
        }
        if (batch.Count > 0) { db.Completions.AddRange(batch); await db.SaveChangesAsync(ct); }
    }
}
```
