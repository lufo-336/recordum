```csharp
using Recordum.Core.Abstractions;
using Recordum.Core.Enums;
using Recordum.Core.Models;

namespace Recordum.Core.Services;

public sealed class IntensityRampResolver : IIntensityRampResolver
{
    public IntensityRamp Resolve(HabitNode node, HabitConfig? config, Goal? activeGoal)
    {
        ArgumentNullException.ThrowIfNull(node);
        if (activeGoal is null || activeGoal.TargetValue <= 0m) return IntensityRamp.Fallback;

        var perDayTarget = NormalizeToPerDay(activeGoal);
        if (perDayTarget is null || perDayTarget <= 0m) return IntensityRamp.Fallback;

        var t = perDayTarget.Value;
        return new IntensityRamp(t * 0.25m, t * 0.5m, t * 0.75m, t);
    }

    private static decimal? NormalizeToPerDay(Goal goal) => goal.PeriodType switch
    {
        GoalPeriodType.Open => goal.TargetValue,
        GoalPeriodType.Rolling when goal.RollingDays is > 0 => goal.TargetValue / goal.RollingDays.Value,
        GoalPeriodType.Fixed when goal.PeriodStart is { } s && goal.PeriodEnd is { } e && e >= s
            => goal.TargetValue / (e.DayNumber - s.DayNumber + 1),
        _ => null,
    };
}
```

```csharp
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Abstractions.Services.Dtos;

namespace Recordum.Data.Services.Aggregators;

public sealed class NaiveDailyStreakAggregator(ICompletionService completions) : IStreakAggregator
{
    public async ValueTask<IReadOnlyList<StreakRun>> GetHistoryAsync(
        Guid habitNodeId, DateOnly fromDate, DateOnly toDate, CancellationToken ct = default)
    {
        var cells = await completions.GetHeatmapAsync(habitNodeId, fromDate, toDate, ct).ConfigureAwait(false);
        var active = cells.Where(c => c.Count > 0).Select(c => c.Date).ToHashSet();
        if (active.Count == 0) return Array.Empty<StreakRun>();

        var ordered = active.OrderBy(d => d.DayNumber).ToList();
        var runs = new List<StreakRun>();
        var runStart = ordered[0];
        var runEnd = ordered[0];
        for (int i = 1; i < ordered.Count; i++)
        {
            if (ordered[i].DayNumber == runEnd.DayNumber + 1)
            {
                runEnd = ordered[i];
            }
            else
            {
                runs.Add(new StreakRun(runStart, runEnd, runEnd.DayNumber - runStart.DayNumber + 1));
                runStart = ordered[i];
                runEnd = ordered[i];
            }
        }
        runs.Add(new StreakRun(runStart, runEnd, runEnd.DayNumber - runStart.DayNumber + 1));
        return runs;
    }
}
```

```csharp
public async ValueTask<IReadOnlyList<DailyValue>> GetDailySumAsync(
    Guid habitNodeId, Guid metricDefinitionId, DateOnly fromDate, DateOnly toDate, CancellationToken ct = default)
{
    var fromUtc = fromDate.ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc);
    var toUtc = toDate.AddDays(1).ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc);
    await using var db = await _factory.CreateDbContextAsync(ct).ConfigureAwait(false);
    var rows = await db.Completions
        .Where(c => c.HabitNodeId == habitNodeId && c.CompletedAt >= fromUtc && c.CompletedAt < toUtc)
        .SelectMany(c => c.Values.Where(v => v.MetricDefinitionId == metricDefinitionId),
                    (c, v) => new { c.CompletedAt, v.ValueNumeric })
        .GroupBy(x => x.CompletedAt.Date)
        .Select(g => new { Date = g.Key, Sum = g.Sum(x => x.ValueNumeric ?? 0m) })
        .ToListAsync(ct).ConfigureAwait(false);
    return rows.Select(r => new DailyValue(DateOnly.FromDateTime(r.Date), r.Sum)).ToList();
}
```

```csharp
using Microsoft.EntityFrameworkCore;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Enums;
using System.Globalization;

namespace Recordum.Data.Services.Queries;

public sealed class TrendQueryService(IDbContextFactory<RecordumDbContext> dbFactory, IMetricDefinitionService metrics) : ITrendQueryService
{
    public async ValueTask<IReadOnlyList<TrendPoint>> GetTrendAsync(
        Guid habitNodeId, DateOnly fromDate, DateOnly toDate, TrendGranularity granularity, CancellationToken ct = default)
    {
        var fromUtc = fromDate.ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc);
        var toUtc = toDate.AddDays(1).ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc);

        await using var db = await dbFactory.CreateDbContextAsync(ct).ConfigureAwait(false);
        var habit = await db.HabitNodes.AsNoTracking().FirstOrDefaultAsync(h => h.Id == habitNodeId, ct).ConfigureAwait(false);
        if (habit is null) return Array.Empty<TrendPoint>();

        var primary = (await metrics.ListByHabitAsync(habitNodeId, ct).ConfigureAwait(false))
            .OrderBy(m => m.SortOrder).FirstOrDefault();

        var completions = await db.Completions.AsNoTracking()
            .Where(c => c.HabitNodeId == habitNodeId && c.CompletedAt >= fromUtc && c.CompletedAt < toUtc)
            .Select(c => new { c.Id, c.CompletedAt, Values = c.Values.Select(v => new { v.MetricDefinitionId, v.ValueNumeric, v.ValueInteger }).ToList() })
            .ToListAsync(ct).ConfigureAwait(false);

        if (completions.Count == 0) return Array.Empty<TrendPoint>();

        IEnumerable<IGrouping<DateOnly, dynamic>> grouped = completions
            .GroupBy(c => BucketKey(DateOnly.FromDateTime(c.CompletedAt), granularity), c => (dynamic)c);

        var points = grouped
            .OrderBy(g => g.Key)
            .Select(g =>
            {
                var rows = g.ToList();
                var count = rows.Count;
                decimal value;
                if (habit.HabitType == HabitType.Action || habit.HabitType == HabitType.Journal || primary is null)
                {
                    value = count;
                }
                else if (habit.HabitType == HabitType.State)
                {
                    var ints = rows.SelectMany(r => ((IEnumerable<dynamic>)r.Values))
                                   .Where(v => (Guid?)v.MetricDefinitionId == primary.Id && v.ValueInteger != null)
                                   .Select(v => (decimal)(int)v.ValueInteger).ToList();
                    value = ints.Count == 0 ? 0m : ints.Sum() / ints.Count;
                }
                else
                {
                    value = rows.SelectMany(r => ((IEnumerable<dynamic>)r.Values))
                               .Where(v => (Guid?)v.MetricDefinitionId == primary.Id && v.ValueNumeric != null)
                               .Sum(v => (decimal)v.ValueNumeric);
                }
                return new TrendPoint(g.Key, value, count);
            })
            .ToList();
        return points;
    }

    private static DateOnly BucketKey(DateOnly date, TrendGranularity g) => g switch
    {
        TrendGranularity.Daily => date,
        TrendGranularity.Weekly => date.AddDays(-((int)date.DayOfWeek + 6) % 7), // ISO week Monday
        _ => date,
    };
}
```

Note: ISO `DayOfWeek` puts Sunday=0; the formula `(d+6) % 7` shifts Monday→0 so subtraction lands on the Monday of the week.

```csharp
using Microsoft.EntityFrameworkCore;
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Services;

namespace Recordum.Data.Services.Queries;

public sealed class HourlyDistributionQueryService(IDbContextFactory<RecordumDbContext> dbFactory, IClock clock) : IHourlyDistributionQueryService
{
    public async ValueTask<IReadOnlyList<int>> GetHourlyAsync(
        Guid habitNodeId, DateOnly fromDate, DateOnly toDate, CancellationToken ct = default)
    {
        var fromUtc = fromDate.ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc);
        var toUtc = toDate.AddDays(1).ToDateTime(TimeOnly.MinValue, DateTimeKind.Utc);

        await using var db = await dbFactory.CreateDbContextAsync(ct).ConfigureAwait(false);
        var timestamps = await db.Completions.AsNoTracking()
            .Where(c => c.HabitNodeId == habitNodeId && !c.IsRetroactive && c.CompletedAt >= fromUtc && c.CompletedAt < toUtc)
            .Select(c => c.CompletedAt)
            .ToListAsync(ct).ConfigureAwait(false);

        var buckets = new int[24];
        foreach (var utc in timestamps)
        {
            var local = TimeZoneInfo.ConvertTimeFromUtc(DateTime.SpecifyKind(utc, DateTimeKind.Utc), clock.LocalTimeZone);
            buckets[local.Hour]++;
        }
        return buckets;
    }
}
```

```csharp
            PeriodLabel = string.Empty;
            DaysLeft = null;
            IsDone = false;
            return;

        Target = goal.TargetValue;

        var toUtc = new DateTimeOffset(to.AddDays(1).ToDateTime(TimeOnly.MinValue), TimeSpan.Zero);

                         .Where(v => v.MetricDefinitionId == metricId)
                         .Sum(v => v.ValueNumeric ?? (v.ValueInteger ?? 0));
            UnitLabel = string.Empty;

            Actual = list.Count;
            UnitLabel = "completions";

        IsDone = Actual >= Target && Target > 0;

    {

        return goal.PeriodType switch
        {
            GoalPeriodType.Fixed => (goal.PeriodStart ?? today, (goal.PeriodEnd ?? today) < today ? goal.PeriodEnd!.Value : today),
            GoalPeriodType.Rolling => (today.AddDays(-((goal.RollingDays ?? 1) - 1)), today),
            _ => (DateOnly.FromDateTime(node.CreatedAt), today),
        };

    {

        var today = DateOnly.FromDateTime(clock.UtcNow.UtcDateTime);
        var d = goal.PeriodEnd.Value.DayNumber - today.DayNumber;
        return d > 0 ? d : null;
    }

        _ => Strings.Detail_GoalCard_Period_AllTime,
    };
```

```csharp
        anchor = anchor.AddDays(-(((int)anchor.DayOfWeek + 6) % 7));

        Dictionary<DateOnly, decimal> values;

            }
            else

                values = (await completions.GetHeatmapAsync(node.Id, anchor, today, ct).ConfigureAwait(true))
                    .ToDictionary(c => c.Date, c => (decimal)c.Count);
            }

                .ToDictionary(c => c.Date, c => (decimal)c.Count);
        }

        var totalCells = Cols * Rows;
```

```csharp
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Services;
```

```csharp
using Recordum.Plugins.Contracts.Events;
using Recordum.Plugins.Contracts.Hosting;

namespace Recordum.WPF.ViewModels.HabitScreen.Detail;

public sealed partial class DetailViewModel : ObservableObject, IDisposable
{
    private readonly IGoalService _goals;
    private readonly IHabitConfigService _configs;
    private readonly List<IDisposable> _subs = [];

    [ObservableProperty] private HabitNode? _node;
    [ObservableProperty] private bool _isGroup;
    [ObservableProperty] private bool _isHabit;

    public GoalProgressCardViewModel Goal { get; }
    public YearHeatmapViewModel YearHeatmap { get; }
    public StreakSummaryViewModel StreakSummary { get; }
    public StreakTimelineViewModel StreakTimeline { get; }
    public TrendChartViewModel Trend { get; }
    public HourlyHistogramViewModel Hourly { get; }

    public DetailViewModel(
        IGoalService goals,
        IStreakAggregator streaks,
        ITrendQueryService trend,
        IHourlyDistributionQueryService hourly,
        ICompletionService completions,
        IMetricDefinitionService metrics,
        IIntensityRampResolver ramp,
        IHabitConfigService configs,
        IClock clock,
        IEventBus bus)
    {
        _goals = goals;
        _configs = configs;
        Goal = new GoalProgressCardViewModel(goals, completions, clock);
        YearHeatmap = new YearHeatmapViewModel(completions, metrics, ramp, clock);
        StreakSummary = new StreakSummaryViewModel(streaks, clock);
        StreakTimeline = new StreakTimelineViewModel(streaks, clock);
        Trend = new TrendChartViewModel(trend, clock);
        Hourly = new HourlyHistogramViewModel(hourly, clock);
        Trend.RangeChanged += async (_, r) => await Hourly.SetRangeAsync(r).ConfigureAwait(true);

        _subs.Add(bus.Subscribe(WellKnownEvents.CompletionRecorded, OnCompletionEvent));
        _subs.Add(bus.Subscribe(WellKnownEvents.CompletionUpdated, OnCompletionEvent));
        _subs.Add(bus.Subscribe(WellKnownEvents.CompletionDeleted, OnCompletionEvent));
        _subs.Add(bus.Subscribe(WellKnownEvents.NoteCreated, OnNoteEvent));
        _subs.Add(bus.Subscribe(WellKnownEvents.NoteUpdated, OnNoteEvent));
        _subs.Add(bus.Subscribe(WellKnownEvents.NoteDeleted, OnNoteEvent));
    }

    public async Task LoadAsync(HabitNode node, CancellationToken ct = default)
    {
        Node = node;
        IsGroup = node.NodeType == NodeType.Group;
        IsHabit = node.NodeType == NodeType.Habit;
        if (IsGroup)
        {
            await Goal.LoadAsync(node, ct).ConfigureAwait(true);
            return;
        }
        var config = await _configs.GetAsync(node.Id, ct).ConfigureAwait(true);
        var activeGoal = await _goals.GetActiveAsync(node.Id, ct).ConfigureAwait(true);
        await Task.WhenAll(
            Goal.LoadAsync(node, ct),
            YearHeatmap.LoadAsync(node, config, activeGoal, ct),
            StreakSummary.LoadAsync(node, ct),
            StreakTimeline.LoadAsync(node, ct),
            Trend.LoadAsync(node, ct),
            Hourly.LoadAsync(node, ct)
        ).ConfigureAwait(true);
    }

    private async void OnCompletionEvent(EventEnvelope envelope)
    {
        if (Node is null) return;
        Guid habitId = envelope.Payload switch
        {
            CompletionRecordedPayload r => r.HabitNodeId,
            CompletionUpdatedPayload u => u.HabitNodeId,
            CompletionDeletedPayload d => d.HabitNodeId,
            _ => Guid.Empty,
        };
        if (habitId != Node.Id) return;
        await LoadAsync(Node).ConfigureAwait(true);
    }

    private async void OnNoteEvent(EventEnvelope envelope)
    {
        if (Node is null || !IsHabit || Node.HabitType != HabitType.Journal) return;
        Guid habitId = envelope.Payload switch
        {
            NoteCreatedPayload r => r.HabitNodeId,
            NoteUpdatedPayload u => u.HabitNodeId,
            NoteDeletedPayload d => d.HabitNodeId,
            _ => Guid.Empty,
        };
        if (habitId != Node.Id) return;
        await Task.WhenAll(Trend.LoadAsync(Node), Goal.LoadAsync(Node)).ConfigureAwait(true);
    }

    public void Dispose()
    {
        foreach (var s in _subs) s.Dispose();
        _subs.Clear();
    }
```

```csharp
using System.Windows;
using System.Windows.Controls;
```

```csharp
using System.Windows.Shapes;
using Recordum.WPF.Resources;
using Recordum.WPF.ViewModels.HabitScreen.Detail;

namespace Recordum.WPF.Controls;

public partial class TrendChartControl : UserControl
{
    public TrendChartControl()
    {
        InitializeComponent();
        DataContextChanged += (_, __) => Redraw();
        Loaded += (_, __) => Redraw();
    }

    private void OnSizeChanged(object sender, SizeChangedEventArgs e) => Redraw();

    private void Redraw()
    {
        LayoutCanvas.Children.Clear();
        EmptyText.Visibility = Visibility.Collapsed;
        if (DataContext is not TrendChartViewModel vm || ActualWidth <= 0 || ActualHeight <= 0) return;
        if (vm.Series.Count == 0 || vm.MaxValue == 0)
        {
            EmptyText.Text = Strings.Detail_Trend_Empty;
            EmptyText.Visibility = Visibility.Visible;
            return;
        }
        double margin = 24, plotW = ActualWidth - 2 * margin, plotH = ActualHeight - 2 * margin;
        if (plotW <= 0 || plotH <= 0) return;
        var stroke = (Brush)FindResource("Brush.Accent");
        var area = stroke.Clone(); area.Opacity = 0.2;

        if (vm.ChartKind == TrendChartKind.Bar)
        {
            double barW = plotW / vm.Series.Count;
            for (int i = 0; i < vm.Series.Count; i++)
            {
                var p = vm.Series[i];
                double h = plotH * (double)(p.Value / vm.MaxValue);
                var rect = new Rectangle { Width = Math.Max(2, barW - 2), Height = h, Fill = stroke };
                Canvas.SetLeft(rect, margin + i * barW);
                Canvas.SetTop(rect, margin + (plotH - h));
                rect.ToolTip = $"{p.Bucket:yyyy-MM-dd} — {p.Count}";
                LayoutCanvas.Children.Add(rect);
            }
        }
        else
        {
            var poly = new Polyline { Stroke = stroke, StrokeThickness = 2 };
            var areaPoly = new Polygon { Fill = area };
            double xStep = vm.Series.Count <= 1 ? plotW : plotW / (vm.Series.Count - 1);
            var points = new PointCollection();
            for (int i = 0; i < vm.Series.Count; i++)
            {
                double y = margin + plotH * (1 - (double)(vm.Series[i].Value / vm.MaxValue));
                double x = margin + i * xStep;
                points.Add(new Point(x, y));
            }
            poly.Points = points;
            areaPoly.Points = new PointCollection(points);
            areaPoly.Points.Insert(0, new Point(margin, margin + plotH));
            areaPoly.Points.Add(new Point(margin + (vm.Series.Count - 1) * xStep, margin + plotH));
            LayoutCanvas.Children.Add(areaPoly);
            LayoutCanvas.Children.Add(poly);
        }
    }
}
```
