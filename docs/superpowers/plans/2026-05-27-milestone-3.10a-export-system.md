# Milestone 3.10a — Export System — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a 5-format export system (PDF/HTML/Markdown/JSON/CSV) for a single habit or the whole library, rendered by a first-party `Recordum.Plugins.ExportEngine` that registers one formatter per format under the `export.format` extension point — the second real consumer of the plugin API.

**Architecture:** A UI-free `IExportFormatter` contract + `ExportModel` DTO live in `Recordum.Plugins.Contracts`. `Recordum.Data` gains `IExportQueryService` (assembles an `ExportModel` from a persisted habit/library + completions/notes/streaks/heatmap) and `IExportJobService` (history). A new `Recordum.Plugins.ExportEngine` assembly (`net10.0`, sole home of QuestPDF + the SkiaSharp heatmap renderer) implements the 5 formatters, registers them through a manifest + `PluginManager` + `ExtensionPointRegistry` under plugin id `io.recordum.export-engine`. The WPF export dialog resolves a formatter through the registry — never referencing the engine directly.

**Tech Stack:** C# / .NET 10, EF Core + SQLite, CommunityToolkit.Mvvm (WPF), QuestPDF (Community), SkiaSharp, xUnit + Xunit.StaFact.

**Spec:** `docs/superpowers/specs/2026-05-27-milestone-3.10a-export-system-design.md` — read it first.

**Schema delta:** ZERO. `export_job`/`export_template` exist since InitialSchema; `schema_version` stays `2.6`. No migration.

**Conventions to honor (from CLAUDE.md):** services `sealed` with primary constructors + `IDbContextFactory`; never `DateTime.UtcNow` (use `IClock`); all user strings in `Strings.resx`/`Strings.it.resx` (English-first, dotted resx keys → underscore designer props); explicit cascade/index in Fluent API (N/A — no schema change); MVVM with `[ObservableProperty]`/`[RelayCommand]`; first-party features register via `IExtensionPoint` (no shortcuts). The repo has `TreatWarningsAsErrors` + `AnalysisLevel=latest-recommended` — keep analyzers happy (`CultureInfo.InvariantCulture` on `ToString`, dispose `IDisposable`, etc.).

**Test command (whole suite):** `dotnet test Recordum.slnx`
**Build command:** `dotnet build Recordum.slnx`

```csharp
using System.Text.Json.Serialization;

namespace Recordum.Plugins.Contracts.Export;

public sealed record ExportLayout(
    [property: JsonPropertyName("metadata")] bool Metadata = true,
    [property: JsonPropertyName("summary")] bool Summary = true,
    [property: JsonPropertyName("heatmap")] bool Heatmap = true,
    [property: JsonPropertyName("completions")] bool Completions = true,
    [property: JsonPropertyName("notes")] bool Notes = true,
    [property: JsonPropertyName("range")] string Range = "all")
{
    private static readonly JsonSerializerOptions Options = new() { DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull };

    public static ExportLayout Parse(string? json)
    {
        if (string.IsNullOrWhiteSpace(json) || json.Trim() == "{}") return new ExportLayout();
        try { return JsonSerializer.Deserialize<ExportLayout>(json, Options) ?? new ExportLayout(); }
        catch (JsonException) { return new ExportLayout(); }
    }

    public string ToJson() => JsonSerializer.Serialize(this, Options);
}
```

```csharp
using Microsoft.EntityFrameworkCore;
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Enums;
using Recordum.Core.Models;
using Recordum.Plugins.Contracts.Export;

namespace Recordum.Data.Services.Queries;

public sealed class ExportQueryService(
    IDbContextFactory<RecordumDbContext> dbFactory,
    IClock clock,
    ICompletionService completions,
    INoteService notes,
    IMetricDefinitionService metrics,
    IGoalService goals,
    IHabitConfigService configs,
    IIntensityRampResolver ramp,
    IStreakAggregator streaks) : IExportQueryService
{
    private const int HeatmapCols = 53;
    private const int HeatmapRows = 7;

    public async ValueTask<ExportModel> BuildAsync(ExportRequest request, CancellationToken ct = default)
    {
        ArgumentNullException.ThrowIfNull(request);
        var generatedAt = clock.UtcNow;

        var habits = new List<ExportHabitModel>();
        if (request.Subject == ExportSubject.SingleHabit)
        {
            await using var db = await dbFactory.CreateDbContextAsync(ct).ConfigureAwait(false);
            var node = request.HabitNodeId is { } id
                ? await db.HabitNodes.AsNoTracking().FirstOrDefaultAsync(n => n.Id == id, ct).ConfigureAwait(false)
                : null;
            if (node is { NodeType: NodeType.Habit })
                habits.Add(await BuildHabitAsync(node, request, ct).ConfigureAwait(false));
        }

        var title = request.Subject == ExportSubject.SingleHabit
            ? habits.FirstOrDefault()?.Name ?? ""
            : "Recordum Library";
        return new ExportModel(request.Subject, title, request.From, request.To, generatedAt, habits);
    }

    private async Task<ExportHabitModel> BuildHabitAsync(HabitNode node, ExportRequest request, CancellationToken ct)
    {
        var fromUtc = new DateTimeOffset(request.From.ToDateTime(TimeOnly.MinValue), TimeSpan.Zero);
        var toUtc = new DateTimeOffset(request.To.ToDateTime(TimeOnly.MaxValue), TimeSpan.Zero);

        var completionRows = request.Layout.Completions
            ? (await completions.ListByHabitAsync(node.Id, fromUtc, toUtc, take: null, ct).ConfigureAwait(false))
                .Select(c => new ExportCompletionRow(
                    new DateTimeOffset(DateTime.SpecifyKind(c.CompletedAt, DateTimeKind.Utc), TimeSpan.Zero),
                    c.ValueDecimal ?? c.ValueInteger ?? c.ValueSeconds, null, c.IsRetroactive))
                .ToList()
            : [];

        var noteRows = request.Layout.Notes
            ? (await notes.ListByHabitAsync(node.Id, ct).ConfigureAwait(false))
                .Select(n => new ExportNoteRow(DateTimeOffset.FromUnixTimeMilliseconds(n.UpdatedAtMs), n.BodyMd))
                .ToList()
            : [];

        var summary = await BuildSummaryAsync(node, request, ct).ConfigureAwait(false);
        var heatmap = request.Layout.Heatmap
            ? await BuildHeatmapAsync(node, ct).ConfigureAwait(false)
            : new List<ExportHeatmapCell>();

        return new ExportHabitModel(node.Id, node.Name, node.HabitType?.ToString() ?? "", node.Color,
            summary, completionRows, noteRows, heatmap);
    }

    private async Task<ExportSummary> BuildSummaryAsync(HabitNode node, ExportRequest request, CancellationToken ct)
    {
        var runs = await streaks.GetHistoryAsync(node.Id, request.From, request.To, ct).ConfigureAwait(false);
        var current = runs.FirstOrDefault(r => r.IsCurrent)?.Length ?? 0;
        var longest = runs.Count == 0 ? 0 : runs.Max(r => r.Length);

        var fromUtc = new DateTimeOffset(request.From.ToDateTime(TimeOnly.MinValue), TimeSpan.Zero);
        var toUtc = new DateTimeOffset(request.To.ToDateTime(TimeOnly.MaxValue), TimeSpan.Zero);
        var total = (await completions.ListByHabitAsync(node.Id, fromUtc, toUtc, take: null, ct).ConfigureAwait(false)).Count;

        var goal = await goals.GetActiveAsync(node.Id, ct).ConfigureAwait(false);
        decimal? target = goal?.TargetValue;
        decimal? actual = null;
        if (goal is not null)
        {
            actual = goal.MetricDefinitionId is { } mid
                ? (await completions.GetDailySumAsync(node.Id, mid, request.From, request.To, ct).ConfigureAwait(false)).Sum(r => r.Sum)
                : total;
        }
        return new ExportSummary(current, longest, total, actual, target);
    }

    private async Task<IReadOnlyList<ExportHeatmapCell>> BuildHeatmapAsync(HabitNode node, CancellationToken ct)
    {
        var today = DateOnly.FromDateTime(clock.UtcNow.UtcDateTime);
        var mondayOfTodayWeek = today.AddDays(-(((int)today.DayOfWeek + 6) % 7));
        var anchor = mondayOfTodayWeek.AddDays(-(HeatmapCols - 1) * HeatmapRows);

        Dictionary<DateOnly, decimal> values;
        if (node.HabitType is HabitType.Counter or HabitType.Resistance or HabitType.State)
        {
            var primary = (await metrics.ListByHabitAsync(node.Id, ct).ConfigureAwait(false))
                .OrderBy(m => m.SortOrder).FirstOrDefault();
            values = primary is not null
                ? (await completions.GetDailySumAsync(node.Id, primary.Id, anchor, today, ct).ConfigureAwait(false)).ToDictionary(r => r.Date, r => r.Sum)
                : (await completions.GetHeatmapAsync(node.Id, anchor, today, ct).ConfigureAwait(false)).ToDictionary(c => c.Date, c => (decimal)c.Count);
        }
        else
        {
            values = (await completions.GetHeatmapAsync(node.Id, anchor, today, ct).ConfigureAwait(false)).ToDictionary(c => c.Date, c => (decimal)c.Count);
        }

        var config = await configs.GetAsync(node.Id, ct).ConfigureAwait(false);
        var goal = await goals.GetActiveAsync(node.Id, ct).ConfigureAwait(false);
        var rampSteps = ramp.Resolve(node, config, goal);

        var cells = new List<ExportHeatmapCell>(HeatmapCols * HeatmapRows);
        for (int i = 0; i < HeatmapCols * HeatmapRows; i++)
        {
            var d = anchor.AddDays(i);
            if (d > today) continue;
            var v = values.GetValueOrDefault(d, 0m);
            int step = v <= 0 ? 0 : v < rampSteps.Step1 ? 1 : v < rampSteps.Step2 ? 2 : v < rampSteps.Step3 ? 3 : 4;
            if (step > 0) cells.Add(new ExportHeatmapCell(i / HeatmapRows, i % HeatmapRows, step));
        }
        return cells;
    }
}
```

Only cells with `step > 0` are emitted (sparse) — empty days are implicit.

```csharp
using Recordum.Plugins.Contracts.Export;
using SkiaSharp;

namespace Recordum.Plugins.ExportEngine;

/// <summary>Headless SkiaSharp renderer of the 53x7 year heatmap to a PNG. 5-step GitHub light ramp.</summary>
public static class HeatmapImageRenderer
{
    private const int Cols = 53, Rows = 7, Cell = 11, Gap = 2, Pad = 6;

    private static readonly SKColor[] Ramp =
    [
        SKColor.Parse("#EBEDF0"), SKColor.Parse("#9BE9A8"), SKColor.Parse("#40C463"),
        SKColor.Parse("#30A14E"), SKColor.Parse("#216E39"),
    ];

    public static byte[] RenderPng(IReadOnlyList<ExportHeatmapCell> cells)
    {
        ArgumentNullException.ThrowIfNull(cells);
        int w = Pad * 2 + Cols * (Cell + Gap);
        int h = Pad * 2 + Rows * (Cell + Gap);

        using var bmp = new SKBitmap(w, h);
        using (var canvas = new SKCanvas(bmp))
        {
            canvas.Clear(SKColors.White);
            using var paint = new SKPaint { IsAntialias = false, Style = SKPaintStyle.Fill };
            // empty grid
            for (int c = 0; c < Cols; c++)
                for (int r = 0; r < Rows; r++)
                    DrawCell(canvas, paint, c, r, Ramp[0]);
            // filled cells
            foreach (var cell in cells)
            {
                if (cell.Week is < 0 or >= Cols || cell.Weekday is < 0 or >= Rows) continue;
                var step = Math.Clamp(cell.Step, 0, Ramp.Length - 1);
                DrawCell(canvas, paint, cell.Week, cell.Weekday, Ramp[step]);
            }
        }
        using var img = SKImage.FromBitmap(bmp);
        using var data = img.Encode(SKEncodedImageFormat.Png, 100);
        return data.ToArray();
    }

    private static void DrawCell(SKCanvas canvas, SKPaint paint, int col, int row, SKColor color)
    {
        paint.Color = color;
        float x = Pad + col * (Cell + Gap);
        float y = Pad + row * (Cell + Gap);
        canvas.DrawRect(x, y, Cell, Cell, paint);
    }
}
```

```csharp
    [ObservableProperty] private ExportFormat _selectedFormat = ExportFormat.Pdf;

    [ObservableProperty] private bool _includeMetadata = true;
    [ObservableProperty] private bool _includeSummary = true;
    [ObservableProperty] private bool _includeHeatmap = true;
    [ObservableProperty] private bool _includeCompletions = true;
    [ObservableProperty] private bool _includeNotes = true;

    [ObservableProperty] private ExportTemplate? _selectedTemplate;

    public ExportDialogViewModel(
        IExportQueryService query, IExportJobService jobs, IExtensionPointRegistry registry,
        ISaveFileDialog save, IExportTemplateService templates, IClock clock)
    {

        ArgumentNullException.ThrowIfNull(jobs);
        ArgumentNullException.ThrowIfNull(registry);
        ArgumentNullException.ThrowIfNull(save);
        ArgumentNullException.ThrowIfNull(templates);
        ArgumentNullException.ThrowIfNull(clock);
        _query = query; _jobs = jobs; _registry = registry; _save = save; _templates = templates; _clock = clock;
    }

    public async Task LoadTemplatesAsync(CancellationToken ct = default)
    {
        Templates.Clear();
        foreach (var t in await _templates.ListAsync(ct).ConfigureAwait(true)) Templates.Add(t);
    }

    partial void OnSelectedTemplateChanged(ExportTemplate? value)
    {
        if (value is null) return;
        var layout = ExportLayout.Parse(value.LayoutJson);
        IncludeMetadata = layout.Metadata;
        IncludeSummary = layout.Summary;
        IncludeHeatmap = layout.Heatmap;
        IncludeCompletions = layout.Completions;
        IncludeNotes = layout.Notes;
        Range = layout.Range;
    }

    public void LoadForHabit(Guid habitId, string name)
    {
        _subject = ExportSubject.SingleHabit;
        _habitId = habitId;
        SubjectLabel = name;

        _habitId = null;
        SubjectLabel = "Library";

    [RelayCommand]
    private async Task ExportAsync(CancellationToken ct)
    {
        var formatter = _registry.GetExtensions(WellKnownExtensionPoints.ExportFormat)
            .OfType<IExportFormatter>()

        "90d" => today.AddDays(-89),
        "1y" => today.AddDays(-364),
        _ => new DateOnly(2000, 1, 1), // "all"
```

```csharp
using System.Windows;
using Microsoft.Extensions.DependencyInjection;
using Recordum.WPF.ViewModels.Export;
using Recordum.WPF.Views;

namespace Recordum.WPF.Services;

public sealed class WpfExportDialogLauncher(IServiceProvider services) : IExportDialogLauncher
{
    public async Task ShowForHabitAsync(Guid habitNodeId, string habitName)
    {
        var vm = services.GetRequiredService<ExportDialogViewModel>();
        await vm.LoadTemplatesAsync().ConfigureAwait(true);
        vm.LoadForHabit(habitNodeId, habitName);
        ShowModal(vm);
    }

    public async Task ShowForLibraryAsync()
    {
        var vm = services.GetRequiredService<ExportDialogViewModel>();
        await vm.LoadTemplatesAsync().ConfigureAwait(true);
        vm.LoadForLibrary();
        ShowModal(vm);
    }

    private static void ShowModal(ExportDialogViewModel vm)
    {
        var dlg = new ExportDialog(vm) { Owner = Application.Current?.MainWindow };
        dlg.ShowDialog();
    }
}
```
