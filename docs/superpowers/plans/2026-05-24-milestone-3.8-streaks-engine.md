# Milestone 3.8 — Streaks Engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace `NaiveDailyStreakAggregator` with `SchedulingAwareStreakAggregator` honoring `ScheduleType × StreakMode × StreakRecovery × StreakPause × HabitConfigHistory × IResetTimeResolver`. Add `IResetTimeResolver` (M3.3 debt). Schema delta v2.6: one column `habit_config.streak_broken_at`.

**Architecture:** Single-pass walker over `[from..to]`. Per day, an `IScheduleEvaluator` decides if scheduled, an `IStreakModeEvaluator` (4 keyed impls) classifies the day as `Counts/Misses/Neutral`, and an `IStreakRecoveryPolicy` (4 keyed impls) decides whether to absorb misses. `HabitConfigTimeline` resolves the active config per day. Stateless; no new tables. `IStreakAggregator` signature preserved → M3.7 ViewModels untouched. `StreakRun` extended non-breaking (`IsCurrent`, `RecoveriesUsed`, `BreakReason?`).

**Tech Stack:** C# / .NET 10, EF Core, CommunityToolkit.Mvvm, xUnit, Xunit.StaFact.

**Source spec:** `docs/superpowers/specs/2026-05-24-milestone-3.8-streaks-engine-design.md`

```csharp
    public async ValueTask<IReadOnlyList<HeatmapCell>> GetHeatmapAsync(
        Guid habitNodeId,
        DateOnly fromDate,
        DateOnly toDate,
        CancellationToken ct = default)
    {
        var fromMs = new DateTimeOffset(fromDate.ToDateTime(TimeOnly.MinValue), TimeSpan.Zero).ToUnixTimeMilliseconds();
        var toMs = new DateTimeOffset(toDate.ToDateTime(TimeOnly.MaxValue), TimeSpan.Zero).ToUnixTimeMilliseconds();
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        var rows = await db.Completions.AsNoTracking()
            .Where(c => c.HabitNodeId == habitNodeId && c.CompletedAtMs >= fromMs && c.CompletedAtMs <= toMs)
            .Select(c => new { c.CompletedAt, c.IsRetroactive })
            .ToListAsync(ct);
        return rows
            .GroupBy(t => DateOnly.FromDateTime(t.CompletedAt))
            .Select(g => new HeatmapCell(g.Key, g.Count(), g.Count(x => x.IsRetroactive)))
            .OrderBy(c => c.Date)
            .ToList();
    }
```

```csharp
using System.Globalization;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Models;

namespace Recordum.Data.Services;

public sealed partial class ResetTimeResolver(
    IDbContextFactory<RecordumDbContext> dbFactory,
    IAppSettingService settings,
    ILogger<ResetTimeResolver> logger) : IResetTimeResolver
{
    private const int MaxDepth = 32;
    private static readonly TimeOnly DefaultReset = new(0, 0);

    [LoggerMessage(EventId = 1, Level = LogLevel.Warning, Message = "Reset-time cascade exceeded depth for {NodeId}; falling back")]
    private partial void LogDepthExceeded(Guid nodeId);

    public async ValueTask<TimeOnly> GetEffectiveResetTimeAsync(Guid habitNodeId, CancellationToken ct = default)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        var current = await db.HabitNodes.AsNoTracking()
            .Where(n => n.Id == habitNodeId)
            .Select(n => new { n.Id, n.ParentId, n.ResetTime })
            .FirstOrDefaultAsync(ct);

        var seen = new HashSet<Guid>();
        var depth = 0;
        while (current is not null)
        {
            if (current.ResetTime is { } t) return t;
            if (!seen.Add(current.Id) || ++depth > MaxDepth)
            {
                LogDepthExceeded(habitNodeId);
                break;
            }
            if (current.ParentId is not { } pid) break;
            current = await db.HabitNodes.AsNoTracking()
                .Where(n => n.Id == pid)
                .Select(n => new { n.Id, n.ParentId, n.ResetTime })
                .FirstOrDefaultAsync(ct);
        }

        var raw = await settings.GetAsync("daily_reset_time", ct);
        if (raw is not null
            && TimeOnly.TryParseExact(raw, "HH:mm", CultureInfo.InvariantCulture, DateTimeStyles.None, out var parsed))
            return parsed;
        return DefaultReset;
    }
}
```

```csharp
using System.Text.Json;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Enums;

namespace Recordum.Data.Services.Evaluators;

public sealed class ScheduleEvaluator : IScheduleEvaluator
{
    public bool IsScheduled(DateOnly date, ScheduledRules rules)
    {
        if (date < rules.HabitCreatedDate) return false;
        return rules.Type switch
        {
            ScheduleType.Daily => true,
            ScheduleType.WeeklyX => true,
            ScheduleType.Custom => true,
            ScheduleType.SpecificDays => MatchesDayMask(date, rules.DaysJson),
            ScheduleType.EveryXDays => rules.EveryXDays is { } x && x > 0
                && (date.DayNumber - rules.HabitCreatedDate.DayNumber) % x == 0,
            ScheduleType.Monthly => MatchesMonthDays(date, rules.MonthlyDaysJson),
            _ => false
        };
    }

    private static bool MatchesDayMask(DateOnly date, string? json)
    {
        if (string.IsNullOrWhiteSpace(json)) return false;
        try
        {
            var days = JsonSerializer.Deserialize<string[]>(json);
            if (days is null || days.Length == 0) return false;
            var key = date.DayOfWeek switch
            {
                DayOfWeek.Monday => "Mon",
                DayOfWeek.Tuesday => "Tue",
                DayOfWeek.Wednesday => "Wed",
                DayOfWeek.Thursday => "Thu",
                DayOfWeek.Friday => "Fri",
                DayOfWeek.Saturday => "Sat",
                DayOfWeek.Sunday => "Sun",
                _ => ""
            };
            return Array.Exists(days, d => string.Equals(d, key, StringComparison.OrdinalIgnoreCase));
        }
        catch (JsonException) { return false; }
    }

    private static bool MatchesMonthDays(DateOnly date, string? json)
    {
        if (string.IsNullOrWhiteSpace(json)) return false;
        try
        {
            var days = JsonSerializer.Deserialize<int[]>(json);
            if (days is null || days.Length == 0) return false;
            return Array.IndexOf(days, date.Day) >= 0;
        }
        catch (JsonException) { return false; }
    }
}
```

```csharp
using Recordum.Core.Models;

namespace Recordum.Data.Services.Aggregators;

internal sealed class HabitConfigTimeline
{
    private readonly HabitConfig _current;
    private readonly HabitConfigHistory[] _history;

    public HabitConfigTimeline(HabitConfig current, IReadOnlyList<HabitConfigHistory> history)
    {
        _current = current;
        _history = history.OrderBy(h => h.ValidFrom).ToArray();
    }

    public HabitConfig ResolveFor(DateOnly date)
    {
        var dt = date.ToDateTime(TimeOnly.MaxValue);
        for (var i = 0; i < _history.Length; i++)
        {
            var h = _history[i];
            var until = h.ValidUntil ?? DateTime.MaxValue;
            if (h.ValidFrom <= dt && dt < until)
                return Project(h);
        }
        return _current;
    }

    private static HabitConfig Project(HabitConfigHistory h) => new()
    {
        Id = Guid.Empty, HabitNodeId = h.HabitNodeId,
        ScheduleType = h.ScheduleType, ScheduleDaysJson = h.ScheduleDaysJson,
        ScheduleWeeklyTarget = h.ScheduleWeeklyTarget, ScheduleEveryXDays = h.ScheduleEveryXDays,
        ScheduleMonthlyDaysJson = h.ScheduleMonthlyDaysJson,
        StreakMode = h.StreakMode, StreakRecovery = h.StreakRecovery,
        StreakToleranceMinutes = h.StreakToleranceMinutes,
        HasIntegratedTimer = h.HasIntegratedTimer, TimerMode = h.TimerMode,
        TimerTargetSeconds = h.TimerTargetSeconds, TimerAutoPrompt = h.TimerAutoPrompt,
        RestTimerSeconds = h.RestTimerSeconds, SessionBaseTimer = h.SessionBaseTimer,
        SessionCountdownSeconds = h.SessionCountdownSeconds,
    };
}
```

```csharp
using Microsoft.Extensions.DependencyInjection;
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Enums;
using Recordum.Core.Models;

namespace Recordum.Data.Services.Aggregators;

public sealed class SchedulingAwareStreakAggregator(
    IHabitNodeService nodes,
    IHabitConfigService configs,
    ICompletionService completions,
    IStreakPauseService pauses,
    IScheduleEvaluator scheduler,
    IClock clock) : IStreakAggregator
{
    public async ValueTask<IReadOnlyList<StreakRun>> GetHistoryAsync(
        Guid habitNodeId, DateOnly fromDate, DateOnly toDate, CancellationToken ct = default)
    {
        if (fromDate > toDate) throw new ArgumentException("fromDate must be <= toDate", nameof(fromDate));

        var habitNode = await nodes.GetAsync(habitNodeId, ct);
        if (habitNode is null || habitNode.NodeType != NodeType.Habit) return Array.Empty<StreakRun>();

        var cfg = await configs.GetAsync(habitNodeId, ct);
        if (cfg is null) return Array.Empty<StreakRun>();
        var history = await configs.ListHistoryAsync(habitNodeId, ct);
        var timeline = new HabitConfigTimeline(cfg, history);
        var habitCreatedDate = DateOnly.FromDateTime(habitNode.CreatedAt);

        var heatmap = await completions.GetHeatmapAsync(habitNodeId, fromDate, toDate, ct);
        var dailyAll = heatmap.ToDictionary(c => c.Date, c => c.Count);
        var dailyLive = heatmap.ToDictionary(c => c.Date, c => c.Count - c.RetroactiveCount);

        var pauseList = await pauses.ListByHabitAsync(habitNodeId, ct);

        var runs = new List<StreakRun>();
        var runStart = (DateOnly?)null; var runEnd = default(DateOnly);
        var length = 0; var recoveriesUsed = 0;

        var today = DateOnly.FromDateTime(clock.UtcNow.UtcDateTime);

        for (var day = fromDate; day <= toDate; day = day.AddDays(1))
        {
            var cfgForDay = timeline.ResolveFor(day);
            var rules = new ScheduledRules(
                cfgForDay.ScheduleType, cfgForDay.ScheduleDaysJson,
                cfgForDay.ScheduleWeeklyTarget, cfgForDay.ScheduleEveryXDays,
                cfgForDay.ScheduleMonthlyDaysJson, habitCreatedDate);

            var isScheduled = scheduler.IsScheduled(day, rules);
            var completedCount = dailyLive.GetValueOrDefault(day, 0);
            var verdict = DailyVerdict(isScheduled, completedCount);

            if (verdict == DayVerdict.Neutral) continue;

            if (verdict == DayVerdict.Counts)
            {
                runStart ??= day;
                runEnd = day;
                length++;
                continue;
            }

            // Misses (None policy: always close).
            if (runStart is { } rs)
            {
                runs.Add(new StreakRun(rs, runEnd, length, IsCurrent: false,
                    RecoveriesUsed: recoveriesUsed, BreakReason: StreakBreakReason.Missed));
                runStart = null; length = 0; recoveriesUsed = 0;
            }
        }

        if (runStart is { } rsFinal)
        {
            var isCurrent = runEnd >= today.AddDays(-1);
            runs.Add(new StreakRun(rsFinal, runEnd, length, isCurrent, recoveriesUsed, null));
        }

        _ = pauseList; // wired in later tasks
        return runs;
    }

    private static DayVerdict DailyVerdict(bool isScheduled, int completedCount)
    {
        if (!isScheduled) return DayVerdict.Neutral;
        return completedCount > 0 ? DayVerdict.Counts : DayVerdict.Misses;
    }
}
```

```csharp
        // ISO week boundaries: Monday..Sunday.
        var dow = ((int)day.DayOfWeek + 6) % 7; // Mon=0
        var weekStart = day.AddDays(-dow);
        var weekEnd = weekStart.AddDays(6);
        var completionsSoFar = 0;
        for (var d = weekStart; d <= day; d = d.AddDays(1))
            if (d >= createdDate) completionsSoFar += dailyLive.GetValueOrDefault(d, 0);
        return new WeeklyPeriodState(target, completionsSoFar, day == weekEnd);
    }
```

```csharp
[Trait("category", "perf")]

    [Fact]
    public async Task OneYearHistory_completes_within_budget()
    {
        await using var f = await TestDbFactory.CreateMigratedAsync();
        var (sp, scope) = f.CreateScopeWithServices();
        var habit = await SeedHelper.SeedHabitAsync(sp, scope, "perf", HabitType.Action);
        var configs = sp.GetRequiredService<IHabitConfigService>();
        var completions = sp.GetRequiredService<ICompletionService>();
        await configs.UpsertAsync(habit.Id, c =>
        {
            c.ScheduleType = ScheduleType.Daily;
            c.StreakMode = StreakMode.Daily;
            c.StreakRecovery = StreakRecovery.None;
        });

        var start = new DateOnly(2025, 5, 1);
        for (var i = 0; i < 365; i++)
        {
            var when = start.AddDays(i).ToDateTime(new TimeOnly(10, 0));
            await completions.RecordAsync(new Completion
            {
                Id = Guid.NewGuid(), HabitNodeId = habit.Id,
                CompletedAt = when, CompletedAtMs = new DateTimeOffset(when, TimeSpan.Zero).ToUnixTimeMilliseconds(),
                IsRetroactive = false,
            });
        }

        var aggr = sp.GetRequiredService<IStreakAggregator>();

        // Warm-up.
        _ = await aggr.GetHistoryAsync(habit.Id, start, start.AddDays(364));

        var samples = new long[50];
        for (var i = 0; i < samples.Length; i++)
        {
            var sw = Stopwatch.StartNew();
            _ = await aggr.GetHistoryAsync(habit.Id, start, start.AddDays(364));
            sw.Stop();
            samples[i] = sw.ElapsedMilliseconds;
        }
        Array.Sort(samples);
        var p95 = samples[(int)(samples.Length * 0.95) - 1];
        Assert.True(p95 < 200, $"p95={p95}ms exceeds 200ms cap for 1-year history (budget 100ms + slack)");
    }
}
```
