# Milestone 3.5 — Completion Recording Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship completion recording for all five habit types (Action / Counter / Resistance / Journal / State) plus session mode, with quick-action icons on Home, a new right-anchored `HabitScreen` (heatmap + recent list + quick-add + timer), an always-visible `TimerStatusBar` at the bottom of `MainWindow`, retroactive entry via heatmap click, and timer recovery via a new `active_timer` table.

**Architecture:** Composition. New `HabitScreenViewModel` hosts four sub-VMs (Heatmap / QuickAdd polymorphic / TimerControl / CompletionList). New singleton `TimerStatusBarViewModel` observes `IActiveTimerService` and renders one row per active timer. Schema delta v2.5 adds `active_timer` (unique per habit). `CompletionService` is extended with list + heatmap queries and emits `CompletionRecorded/Updated/Deleted` events via `IEventBus`. Per-row quick-action icon on Home delegates to a strategy `IQuickActionHandler` resolved by `HabitType`. Per-field commands persist via existing Core services.

**Tech Stack:** C# / .NET 10, WPF + `CommunityToolkit.Mvvm`, EF Core 9 + SQLCipher, xUnit + `FluentAssertions`.

**Spec:** `docs/superpowers/specs/2026-05-21-milestone-3.5-completion-recording-design.md` is authoritative on conflicts with this plan.

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;
using Recordum.Core.Enums;
using Recordum.Core.Models;

namespace Recordum.Data.Configurations;

internal sealed class ActiveTimerConfiguration : IEntityTypeConfiguration<ActiveTimer>
{
    public void Configure(EntityTypeBuilder<ActiveTimer> b)
    {
        b.ToTable("active_timer");
        b.HasKey(t => t.Id);

        b.Property(t => t.TimerMode)
            .HasConversion<string>()
            .HasMaxLength(16)
            .IsRequired();

        b.Property(t => t.StartedAtMs).IsRequired();
        b.Property(t => t.AccumulatedPausedMs).IsRequired().HasDefaultValue(0L);

        b.HasOne(t => t.HabitNode)
            .WithMany()
            .HasForeignKey(t => t.HabitNodeId)
            .OnDelete(DeleteBehavior.Cascade);

        b.HasIndex(t => t.HabitNodeId).IsUnique().HasDatabaseName("ux_active_timer_habit_node_id");

        b.ToTable(tb => tb.HasCheckConstraint(
            "ck_active_timer_mode",
            "timer_mode IN ('Stopwatch','Countdown','Primary','Session')"));
    }
}

using Microsoft.EntityFrameworkCore;
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Enums;
using Recordum.Core.Errors;
using Recordum.Core.Models;

namespace Recordum.Data.Services;

public sealed class ActiveTimerService(
    IDbContextFactory<RecordumDbContext> dbFactory,
    IClock clock,
    IPluginContext plugin,
    ICompletionService completions) : IActiveTimerService
{
    public async ValueTask<ActiveTimer> StartAsync(Guid habitNodeId, TimerMode mode, CancellationToken ct = default)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        if (await db.ActiveTimers.AnyAsync(t => t.HabitNodeId == habitNodeId, ct))
            throw new TimerAlreadyActiveException(habitNodeId);

        var nowMs = clock.UtcNow.ToUnixTimeMilliseconds();
        var t = new ActiveTimer
        {
            Id = Guid.NewGuid(),
            HabitNodeId = habitNodeId,
            StartedAtMs = nowMs,
            TimerMode = mode,
            SegmentIndex = mode == TimerMode.Session ? 0 : null,
        };
        db.ActiveTimers.Add(t);
        await db.SaveChangesAsync(ct);
        _ = plugin; // suppresses unused — plugin id is stamped by interceptor
        return t;
    }

    public ValueTask ResumeAsync(Guid habitNodeId, CancellationToken ct = default) => throw new NotImplementedException();

    await using var db = await dbFactory.CreateDbContextAsync(ct);
    var q = db.Completions.AsNoTracking().Where(c => c.HabitNodeId == habitNodeId);
    if (from is { } f) { var fMs = f.ToUnixTimeMilliseconds(); q = q.Where(c => c.CompletedAtMs >= fMs); }
    if (to is { } t) { var tMs = t.ToUnixTimeMilliseconds(); q = q.Where(c => c.CompletedAtMs <= tMs); }

public async ValueTask<IReadOnlyList<HeatmapCell>> GetHeatmapAsync(Guid habitNodeId, DateOnly fromDate, DateOnly toDate, CancellationToken ct = default)

public sealed partial class TimerControlViewModel(IActiveTimerService timers, IClock clock) : ObservableObject
{
    private HabitNode? _node;
    private HabitConfig? _cfg;

    [RelayCommand]
    private async Task Start()
