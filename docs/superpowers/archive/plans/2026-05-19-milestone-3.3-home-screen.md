# Milestone 3.3 — Home Screen Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the first user-facing UI of Recordum — a single `MainWindow` with Active/Archived/Trash tabs, filterable habit tree (4 px accent + 7-day inline heatmap), pin model (`PinScope`), drag-and-drop reorder, inline rename, context menu, system/light/dark theming, en+it localization — fully wired to the Milestone 3.2 data layer.

**Architecture:** Schema delta v2.4 (replace `IsPinned : bool` with `PinScope : enum None|Local|Global`, extend `HomeTreeRow` projection with `PinScope` / `ChildCount` / `IsArchived` / `CompletionsLast7Days[7]`). WPF MVVM with `CommunityToolkit.Mvvm`. ViewModels load via `IHabitTreeQueryService`, mutate via existing `IHabitNodeService` + new `SetPinScopeAsync` / `SetArchivedAsync`. Theming via new `IThemeService` (Core) + WPF impl with `SystemEvents.UserPreferenceChanged` watcher and `AppSetting` persistence. Drag-and-drop and inline rename implemented as `Behaviors/` attached behaviors — no external library.

**Tech Stack:** C# / .NET 10, WPF, `CommunityToolkit.Mvvm` (already referenced by `Recordum.WPF.csproj`), EF Core 9 + SQLCipher (already wired), xUnit + `FluentAssertions`.

**Spec:** `docs/superpowers/archive/specs/2026-05-19-milestone-3.3-home-screen-design.md` is authoritative when it conflicts with this plan.

Replace the body of `GetHomeTreeAsync` in `HabitTreeQueryService.cs`. The new projection:

```csharp
public async ValueTask<IReadOnlyList<HomeTreeRow>> GetHomeTreeAsync(CancellationToken ct = default)
{
    await using var db = await _dbFactory.CreateDbContextAsync(ct);
    var today = _clock.UtcNow.UtcDateTime.Date;
    var startMs = new DateTimeOffset(today.AddDays(-6), TimeSpan.Zero).ToUnixTimeMilliseconds();
    var endMs = new DateTimeOffset(today.AddDays(1), TimeSpan.Zero).ToUnixTimeMilliseconds();

    // Pre-aggregate per (habit_node_id, day_offset) for the 7-day window.
    var dayBuckets = await db.Completions
        .AsNoTracking()
        .Where(c => c.CompletedAtMs >= startMs && c.CompletedAtMs < endMs)
        .GroupBy(c => new
        {
            c.HabitNodeId,
            DayOffset = (int)((c.CompletedAtMs - startMs) / 86_400_000),
        })
        .Select(g => new { g.Key.HabitNodeId, g.Key.DayOffset, Count = g.Count() })
        .ToListAsync(ct);

    var bucketsByHabit = dayBuckets
        .GroupBy(b => b.HabitNodeId)
        .ToDictionary(g => g.Key, g =>
        {
            var arr = new int[7];
            foreach (var b in g) if (b.DayOffset is >= 0 and < 7) arr[b.DayOffset] = b.Count;
            return ImmutableArray.Create(arr);
        });

    var emptyHeatmap = ImmutableArray.Create(new int[7]);
    var todayMs = new DateTimeOffset(today, TimeSpan.Zero).ToUnixTimeMilliseconds();
    var todayEndMs = todayMs + 86_400_000;

    var raw = await db.HabitNodes
        .IgnoreQueryFilters() // we need archived AND active in one query; tab predicate is upstream
        .AsNoTracking()
        .Select(n => new
        {
            n.Id,
            n.ParentId,
            n.Name,
            n.Color,
            n.NodeType,
            n.HabitType,
            n.PinScope,
            n.IsArchived,
            n.DeletedAt,
            LastCompletionAt = db.Completions
                .Where(c => c.HabitNodeId == n.Id)
                .OrderByDescending(c => c.CompletedAtMs)
                .Select(c => (DateTime?)c.CompletedAt)
                .FirstOrDefault(),
            IsCompletedToday = db.Completions.Any(c => c.HabitNodeId == n.Id && c.CompletedAtMs >= todayMs && c.CompletedAtMs < todayEndMs),
            ChildCount = db.HabitNodes.Count(child => child.ParentId == n.Id && child.DeletedAt == null),
        })
        .ToListAsync(ct);

    return raw.Select(r => new HomeTreeRow(
        HabitNodeId: r.Id,
        ParentId: r.ParentId,
        Name: r.Name,
        ColorHex: r.Color,
        HabitType: r.HabitType,
        LastCompletionAt: r.LastCompletionAt,
        CurrentStreakCount: 0, // TODO Milestone 3.8
        IsCompletedToday: r.IsCompletedToday,
        PinScope: r.PinScope,
        ChildCount: r.ChildCount,
        IsArchived: r.IsArchived,
        CompletionsLast7Days: bucketsByHabit.TryGetValue(r.Id, out var arr) ? arr : emptyHeatmap))
        .ToList();
}
```

Notes for the engineer:
- `IgnoreQueryFilters()` is used so the caller (`HomeViewModel`) can show Trash/Archived; the existing 3.2 service applied the soft-delete filter implicitly — the change of contract is intentional and called out in the spec §4.4. using Microsoft.EntityFrameworkCore;
```csharp
using Microsoft.Extensions.Logging;
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Models;

namespace Recordum.Data.Services;

public sealed partial class AppSettingService(
    IDbContextFactory<RecordumDbContext> dbFactory,
    IClock clock,
    ILogger<AppSettingService> logger) : IAppSettingService
{
    [LoggerMessage(EventId = 1, Level = LogLevel.Debug, Message = "AppSetting {Key} set")]
    private partial void LogSet(string key);

    public async ValueTask<string?> GetAsync(string key, CancellationToken ct = default)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(key);
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        var row = await db.AppSettings.AsNoTracking().FirstOrDefaultAsync(s => s.Key == key, ct);
        return row?.Value;
    }

    public async ValueTask SetAsync(string key, string value, CancellationToken ct = default)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(key);
        ArgumentNullException.ThrowIfNull(value);
        await using var db = await dbFactory.CreateDbContextAsync(ct);
        var row = await db.AppSettings.FirstOrDefaultAsync(s => s.Key == key, ct);
        if (row is null)
        {
            db.AppSettings.Add(new AppSetting { Key = key, Value = value, UpdatedAt = clock.UtcNow.UtcDateTime });
        }
        else
        {
            row.Value = value;
            row.UpdatedAt = clock.UtcNow.UtcDateTime;
        }
        await db.SaveChangesAsync(ct);
        LogSet(key);
    }

    public async ValueTask<T?> GetEnumAsync<T>(string key, CancellationToken ct = default) where T : struct, Enum
    {
        var raw = await GetAsync(key, ct);
        if (raw is null) return null;
        return Enum.TryParse<T>(raw, ignoreCase: true, out var parsed) ? parsed : null;
    }

    public ValueTask SetEnumAsync<T>(string key, T value, CancellationToken ct = default) where T : struct, Enum
        => SetAsync(key, value.ToString(), ct);
```

```csharp
// src/Recordum.WPF/Theming/ThemeService.cs
using Microsoft.Extensions.Logging;
using Microsoft.Win32;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Abstractions.Theming;
using Recordum.Core.Enums;

namespace Recordum.WPF.Theming;

public sealed partial class ThemeService : IThemeService, IDisposable
{
    private const string PersistenceKey = "theme.mode";
    private readonly IAppSettingService _settings;
    private readonly ILogger<ThemeService> _logger;
    private readonly Func<bool> _systemIsLight;
    private ThemeMode _mode;
    private ThemeMode _effective;
    private bool _disposed;

    public ThemeMode CurrentMode => _mode;
    public ThemeMode EffectiveTheme => _effective;
    public event EventHandler<ThemeMode>? EffectiveThemeChanged;

    public ThemeService(IAppSettingService settings, ILogger<ThemeService> logger)
        : this(settings, logger, ReadSystemIsLight) { }

    // Test-friendly ctor
    internal ThemeService(IAppSettingService settings, ILogger<ThemeService> logger, Func<bool> systemIsLight)
    {
        _settings = settings;
        _logger = logger;
        _systemIsLight = systemIsLight;
    }

    public async ValueTask InitializeAsync(CancellationToken ct = default)
    {
        _mode = (await _settings.GetEnumAsync<ThemeMode>(PersistenceKey, ct)) ?? ThemeMode.System;
        RecomputeEffective(raiseEvent: false);
        SystemEvents.UserPreferenceChanged += OnUserPrefChanged;
    }

    public async ValueTask SetModeAsync(ThemeMode mode, CancellationToken ct = default)
    {
        if (_mode == mode) return;
        _mode = mode;
        await _settings.SetEnumAsync(PersistenceKey, mode, ct);
        RecomputeEffective(raiseEvent: true);
    }

    internal void OnSystemPreferenceChanged() => RecomputeEffective(raiseEvent: true);

    private void OnUserPrefChanged(object? sender, UserPreferenceChangedEventArgs e)
    {
        if (e.Category == UserPreferenceCategory.General) RecomputeEffective(raiseEvent: true);
    }

    private void RecomputeEffective(bool raiseEvent)
    {
        var next = _mode switch
        {
            ThemeMode.Light => ThemeMode.Light,
            ThemeMode.Dark => ThemeMode.Dark,
            _ => _systemIsLight() ? ThemeMode.Light : ThemeMode.Dark,
        };
        if (next == _effective) return;
        _effective = next;
        if (raiseEvent) EffectiveThemeChanged?.Invoke(this, next);
    }

    private static bool ReadSystemIsLight()
    {
        try
        {
            using var rk = Registry.CurrentUser.OpenSubKey(
                @"Software\Microsoft\Windows\CurrentVersion\Themes\Personalize");
            var val = rk?.GetValue("AppsUseLightTheme");
            return val is int i ? i == 1 : true;
        }
        catch
        {
            return true;
        }
    }

    public void Dispose()
    {
        if (_disposed) return;
        SystemEvents.UserPreferenceChanged -= OnUserPrefChanged;
        _disposed = true;
    }
}
```

```csharp
namespace Recordum.WPF.ViewModels;

public sealed partial class HomeRowViewModel : ObservableObject
{
    private readonly IHabitNodeService _nodes;
    private readonly IPurgeService _purge;
    private string _originalName;

    public Guid HabitNodeId { get; }
    public Guid? ParentId { get; }
    public string ColorHex { get; }
    public HabitType? HabitType { get; }
    public bool IsGroup => HabitType is null;
    public int ChildCount { get; }
    public ImmutableArray<int> CompletionsLast7Days { get; }
    public DateTime? LastCompletionAt { get; }
    public bool IsCompletedToday { get; }
    public int IndentLevel { get; }

    [ObservableProperty] private string _name;
    [ObservableProperty] private PinScope _pinScope;
    [ObservableProperty] private bool _isArchived;
    [ObservableProperty] private bool _isExpanded = true;
    [ObservableProperty] private bool _isRenaming;
    [ObservableProperty] private string _pendingName = string.Empty;

    public HomeRowViewModel(HomeTreeRow row, int indentLevel, IHabitNodeService nodes, IPurgeService purge)
    {
        _nodes = nodes;
        _purge = purge;
        HabitNodeId = row.HabitNodeId;
        ParentId = row.ParentId;
        ColorHex = row.ColorHex;
        HabitType = row.HabitType;
        ChildCount = row.ChildCount;
        CompletionsLast7Days = row.CompletionsLast7Days;
        LastCompletionAt = row.LastCompletionAt;
        IsCompletedToday = row.IsCompletedToday;
        IndentLevel = indentLevel;
        _name = row.Name;
        _originalName = row.Name;
        _pinScope = row.PinScope;
        _isArchived = row.IsArchived;
    }

    [RelayCommand]
    private void BeginRename()
    {
        PendingName = Name;
        IsRenaming = true;
    }

    [RelayCommand]
    private async Task CommitRenameAsync()
    {
        if (!IsRenaming) return;
        var newName = PendingName?.Trim();
        if (string.IsNullOrEmpty(newName) || newName == _originalName)
        {
            IsRenaming = false;
            return;
        }
        try
        {
            await _nodes.UpdateAsync(HabitNodeId, n => n.Name = newName);
            _originalName = newName;
            Name = newName;
        }
        catch
        {
            Name = _originalName;
        }
        finally
        {
            IsRenaming = false;
        }
    }

    [RelayCommand]
    private void CancelRename()
    {
        IsRenaming = false;
        PendingName = string.Empty;
    }

    [RelayCommand]
    private async Task SetPinScopeAsync(PinScope scope)
    {
        if (PinScope == scope) return;
        await _nodes.SetPinScopeAsync(HabitNodeId, scope);
        PinScope = scope;
    }

    [RelayCommand]
    private async Task ArchiveAsync()
    {
        await _nodes.SetArchivedAsync(HabitNodeId, true);
        IsArchived = true;
    }

    [RelayCommand]
    private async Task UnarchiveAsync()
    {
        await _nodes.SetArchivedAsync(HabitNodeId, false);
        IsArchived = false;
    }

    [RelayCommand]
    private Task SoftDeleteAsync() => _nodes.SoftDeleteAsync(HabitNodeId).AsTask();

    [RelayCommand]
    private Task RestoreAsync() => _nodes.RestoreAsync(HabitNodeId).AsTask();

    [RelayCommand]
    private Task PurgeAsync() => _purge.PurgeSubtreeAsync(HabitNodeId).AsTask();

    public Task MoveToAsync(Guid? newParent) => _nodes.MoveAsync(HabitNodeId, newParent).AsTask();
```

```csharp
            HomeTab.Archived => _lastFetched.Where(r => r.IsArchived),
```

```csharp
private IEnumerable<(HomeTreeRow, int)> OrderHierarchically(IReadOnlyList<HomeTreeRow> rows)
{
    if (ActiveTab != HomeTab.Active)
    {
        foreach (var r in ApplySort(rows)) yield return (r, 0);
        yield break;
    }

    var byParent = rows.ToLookup(r => r.ParentId);
    // Pin layers operate at root level only — descendants follow their parent's position.
    var roots = ApplySort(byParent[null].ToList())
        .OrderByDescending(r => r.PinScope == PinScope.Global)
        .ThenByDescending(r => r.PinScope == PinScope.Local) // Local before plain at root
        .ToList();

    foreach (var root in roots)
        foreach (var pair in WalkSubtree(root, 0, byParent))
            yield return pair;
}

private IEnumerable<(HomeTreeRow, int)> WalkSubtree(HomeTreeRow node, int level, ILookup<Guid?, HomeTreeRow> byParent)
{
    yield return (node, level);
    var children = byParent[node.HabitNodeId].ToList();
    var sortedChildren = ApplySort(children)
        .OrderByDescending(c => c.PinScope == PinScope.Local) // Local pins float to top of parent
        .ToList();
    foreach (var c in sortedChildren)
        foreach (var pair in WalkSubtree(c, level + 1, byParent))
            yield return pair;
}

private IEnumerable<HomeTreeRow> ApplySort(IReadOnlyList<HomeTreeRow> rows) => SortOption switch
{
    HomeSort.Name => rows.OrderBy(r => r.Name, StringComparer.CurrentCultureIgnoreCase),
    HomeSort.LastCompletion => rows.OrderByDescending(r => r.LastCompletionAt ?? DateTime.MinValue),
    _ => rows,
};
```

```csharp
using System.Collections.Immutable;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;

namespace Recordum.WPF.Views;

public partial class HeatmapStrip : UserControl
{
    public static readonly DependencyProperty CountsProperty =
        DependencyProperty.Register(nameof(Counts), typeof(ImmutableArray<int>), typeof(HeatmapStrip),
            new PropertyMetadata(default(ImmutableArray<int>), OnChanged));
    public static readonly DependencyProperty AccentColorProperty =
        DependencyProperty.Register(nameof(AccentColor), typeof(string), typeof(HeatmapStrip),
            new PropertyMetadata("#3B82F6", OnChanged));

    public ImmutableArray<int> Counts { get => (ImmutableArray<int>)GetValue(CountsProperty); set => SetValue(CountsProperty, value); }
    public string AccentColor { get => (string)GetValue(AccentColorProperty); set => SetValue(AccentColorProperty, value); }

    public HeatmapStrip() { InitializeComponent(); }

    private static void OnChanged(DependencyObject d, DependencyPropertyChangedEventArgs e) => ((HeatmapStrip)d).Rebuild();

    private void Rebuild()
    {
        if (Counts.IsDefault || Counts.Length == 0) { Cells.ItemsSource = null; return; }
        var color = (Color)ColorConverter.ConvertFromString(AccentColor);
        var items = new List<object>();
        for (var i = 0; i < Counts.Length; i++)
        {
            var step = Counts[i] switch { 0 => 0, 1 => 1, 2 => 2, 3 => 3, _ => 4 };
            var alpha = step switch { 0 => (byte)32, 1 => (byte)96, 2 => (byte)160, 3 => (byte)208, _ => (byte)255 };
            var brush = new SolidColorBrush(Color.FromArgb(alpha, color.R, color.G, color.B));
            brush.Freeze();
            items.Add(new { Brush = brush, BorderThickness = i == Counts.Length - 1 ? new Thickness(1) : new Thickness(0) });
        }
        Cells.ItemsSource = items;
    }
}
```

```csharp
// src/Recordum.WPF/Behaviors/InlineRenameBehavior.cs
public static class InlineRenameBehavior
{
    public static readonly DependencyProperty AttachProperty = DependencyProperty.RegisterAttached(
        "Attach", typeof(bool), typeof(InlineRenameBehavior),
        new PropertyMetadata(false, OnAttachChanged));

    public static void SetAttach(DependencyObject d, bool v) => d.SetValue(AttachProperty, v);
    public static bool GetAttach(DependencyObject d) => (bool)d.GetValue(AttachProperty);

    private static void OnAttachChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
    {
        if (d is not TextBox tb) return;
        if ((bool)e.NewValue)
        {
            tb.KeyDown += OnKey;
            tb.LostFocus += OnLostFocus;
            tb.Loaded += OnLoaded;
        }
        else
        {
            tb.KeyDown -= OnKey;
            tb.LostFocus -= OnLostFocus;
            tb.Loaded -= OnLoaded;
        }
    }

    private static void OnLoaded(object? s, RoutedEventArgs e)
    {
        if (s is TextBox tb) { tb.Focus(); tb.SelectAll(); }
    }

    private static void OnKey(object? s, KeyEventArgs e)
    {
        if (s is not TextBox tb || tb.DataContext is not HomeRowViewModel vm) return;
        switch (e.Key)
        {
            case Key.Enter: vm.CommitRenameCommand.Execute(null); e.Handled = true; break;
            case Key.Escape: vm.CancelRenameCommand.Execute(null); e.Handled = true; break;
        }
    }

    private static void OnLostFocus(object? s, RoutedEventArgs e)
    {
        if (s is TextBox tb && tb.DataContext is HomeRowViewModel vm && vm.IsRenaming)
            vm.CommitRenameCommand.Execute(null);
    }
}
```

```csharp
// src/Recordum.WPF/Behaviors/TreeDragDropBehavior.cs
public static class TreeDragDropBehavior
{
    public enum DropZone { Before, Onto, After }

    public static readonly DependencyProperty EnabledProperty = DependencyProperty.RegisterAttached(
        "Enabled", typeof(bool), typeof(TreeDragDropBehavior),
        new PropertyMetadata(false, OnEnabledChanged));

    public static bool GetEnabled(DependencyObject d) => (bool)d.GetValue(EnabledProperty);
    public static void SetEnabled(DependencyObject d, bool v) => d.SetValue(EnabledProperty, v);

    private static void OnEnabledChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
    {
        if (d is not ItemsControl ic) return;
        if ((bool)e.NewValue)
        {
            ic.PreviewMouseLeftButtonDown += OnDown;
            ic.PreviewMouseMove += OnMove;
            ic.Drop += OnDrop;
            ic.AllowDrop = true;
        }
        else
        {
            ic.PreviewMouseLeftButtonDown -= OnDown;
            ic.PreviewMouseMove -= OnMove;
            ic.Drop -= OnDrop;
            ic.AllowDrop = false;
        }
    }

    internal static DropZone ResolveZone(double relative) =>
        relative < 0.4 ? DropZone.Before :
        relative < 0.7 ? DropZone.Onto : DropZone.After;

    private static Point _start;
    private static bool _armed;

    private static void OnDown(object? s, MouseButtonEventArgs e)
    {
        _start = e.GetPosition((IInputElement?)s);
        _armed = true;
    }

    private static void OnMove(object? s, MouseEventArgs e)
    {
        if (!_armed || e.LeftButton != MouseButtonState.Pressed) return;
        var p = e.GetPosition((IInputElement?)s);
        if ((p - _start).Length < 5) return;
        if (e.OriginalSource is FrameworkElement fe && fe.DataContext is HomeRowViewModel src)
        {
            _armed = false;
            DragDrop.DoDragDrop(fe, src, DragDropEffects.Move);
        }
    }

    private static async void OnDrop(object? s, DragEventArgs e)
    {
        if (s is not ItemsControl ic) return;
        if (e.Data.GetData(typeof(HomeRowViewModel)) is not HomeRowViewModel src) return;
        if (e.OriginalSource is not FrameworkElement fe || fe.DataContext is not HomeRowViewModel target) return;
        if (src == target) return;

        var p = e.GetPosition(fe);
        var zone = ResolveZone(p.Y / Math.Max(1, fe.ActualHeight));
        Guid? newParent = zone == DropZone.Onto && target.IsGroup
            ? target.HabitNodeId
            : target.ParentId; // Before/After = sibling
        try { await src.MoveToAsync(newParent); }
        catch (Recordum.Core.Errors.CyclicHabitNodeMoveException) { return; }
        if (ic.DataContext is HomeViewModel home) await home.LoadCommand.ExecuteAsync(null);
    }
}
```

The behavior intentionally does not micromanage `SortOrder` for in-parent reordering in 3.3; that's deferred to a follow-up (post-3.3) since `IHabitNodeService.MoveAsync` currently sets `ParentId` only. The plan acknowledges this — see Phase N risks.
