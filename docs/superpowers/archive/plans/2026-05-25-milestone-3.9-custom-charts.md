# Milestone 3.9 — Custom Charts + Chart Engine Pilot — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship user-defined custom charts in a top-level "Charts" workspace, rendered by a Chart Engine that is the first real consumer of the plugin API (`chart.renderer` extension point), and migrate the four M3.7 detail charts to LiveCharts2.

**Architecture:** A new UI-agnostic `IChartRenderer` contract + `ChartRenderModel` DTO live in `Recordum.Plugins.Contracts`. `Recordum.Data` gains `IChartQueryService` that turns a persisted `Chart` + its `ChartSeries` + completion data into a `ChartRenderModel`. A new `Recordum.Plugins.ChartEngine` assembly (the sole home of LiveCharts2 code) implements `IChartRenderer`, registers itself through a manifest + `PluginManager` + `ExtensionPointRegistry` under plugin id `io.recordum.chart-engine`, and exposes view builders for the detail charts. The WPF Charts workspace resolves the renderer through the registry — never referencing the engine directly for chart rendering.

**Tech Stack:** C# / .NET 10, EF Core + SQLite, CommunityToolkit.Mvvm (WPF), LiveCharts2 (`LiveChartsCore.SkiaSharpView.WPF`, MIT), xUnit + Xunit.StaFact.

**Spec:** `docs/superpowers/archive/specs/2026-05-25-milestone-3.9-custom-charts-design.md` — read it first.

**Schema delta:** ZERO. `chart`/`chart_series` exist since InitialSchema; `Chart.ConfigJson` holds per-chart config. `schema_version` stays `2.6`. No migration in this milestone.

**Conventions to honor (from CLAUDE.md):** services `sealed` with primary constructors + `IDbContextFactory`; never `DateTime.UtcNow` (use `IClock`); all user strings in `Strings.resx`/`Strings.it.resx` (English-first); explicit cascade/index in Fluent API (N/A here — no schema change); MVVM with `[ObservableProperty]`/`[RelayCommand]`.

**Test command (whole suite):** `dotnet test Recordum.slnx`
**Build command:** `dotnet build Recordum.slnx`

namespace Recordum.Plugins.Contracts.Charts;

public static class ChartKindParser
{
    public static string ToToken(ChartKind kind) => kind switch
    {
        ChartKind.Line => "line",
        ChartKind.Bar => "bar",
        ChartKind.Area => "area",
        ChartKind.StackedBar => "stacked-bar",
        ChartKind.Scatter => "scatter",
        ChartKind.Pie => "pie",
        _ => "line",
    };

    public static ChartKind Parse(string? token) => token?.Trim().ToLowerInvariant() switch
    {
        "line" => ChartKind.Line,
        "bar" => ChartKind.Bar,
        "area" => ChartKind.Area,
        "stacked-bar" or "stackedbar" => ChartKind.StackedBar,
        "scatter" => ChartKind.Scatter,
        "pie" => ChartKind.Pie,
        _ => ChartKind.Line,
    };
}

using System.Text.Json;
using System.Text.Json.Serialization;

namespace Recordum.Plugins.Contracts.Charts;

public sealed record ChartConfig(
    [property: JsonPropertyName("range")] string Range = "90d",
    [property: JsonPropertyName("from")] string? FromIso = null,
    [property: JsonPropertyName("to")] string? ToIso = null,
    [property: JsonPropertyName("bucket")] string Bucket = "daily",
    [property: JsonPropertyName("show_legend")] bool ShowLegend = true,
    [property: JsonPropertyName("x_axis_label")] string? XAxisLabel = null,
    [property: JsonPropertyName("y_axis_label")] string? YAxisLabel = null)
{
    private static readonly JsonSerializerOptions Options = new() { DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull };

    public static ChartConfig Parse(string? json)
    {
        if (string.IsNullOrWhiteSpace(json) || json.Trim() == "{}") return new ChartConfig();
        try { return JsonSerializer.Deserialize<ChartConfig>(json, Options) ?? new ChartConfig(); }
        catch (JsonException) { return new ChartConfig(); }
    }

    public string ToJson() => JsonSerializer.Serialize(this, Options);
}

public interface IChartQueryService

    ValueTask<ChartRenderModel> BuildModelAsync(Guid chartId, bool isDark, CancellationToken ct = default);

`ChartQueryService.cs`:

```csharp
using System.Globalization;
using Microsoft.EntityFrameworkCore;
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Services;
using Recordum.Plugins.Contracts.Charts;

namespace Recordum.Data.Services.Queries;

public sealed class ChartQueryService(
    IDbContextFactory<RecordumDbContext> dbFactory,
    IClock clock,
    ITrendQueryService trends,
    ICompletionService completions,
    IMetricDefinitionService metrics) : IChartQueryService
{
    private const string DarkText = "#E6E6E6";
    private const string DarkGrid = "#333333";
    private const string LightText = "#202020";
    private const string LightGrid = "#E0E0E0";

    public async ValueTask<ChartRenderModel> BuildModelAsync(Guid chartId, bool isDark, CancellationToken ct = default)
    {
        var theme = new ChartThemeModel(isDark, isDark ? DarkText : LightText, isDark ? DarkGrid : LightGrid);

        await using var db = await dbFactory.CreateDbContextAsync(ct).ConfigureAwait(false);
        var chart = await db.Charts.AsNoTracking()
            .Include(c => c.Series)
            .FirstOrDefaultAsync(c => c.Id == chartId, ct).ConfigureAwait(false);
        if (chart is null)
            return new ChartRenderModel(ChartKind.Line, null, [], null, null, theme);

        var kind = ChartKindParser.Parse(chart.ChartType);
        var config = ChartConfig.Parse(chart.ConfigJson);
        var (from, to) = ResolveRange(config);
        var granularity = config.Bucket.Equals("weekly", StringComparison.OrdinalIgnoreCase)
            ? TrendGranularity.Weekly : TrendGranularity.Daily;

        var series = new List<ChartSeriesModel>();
        foreach (var s in chart.Series.OrderBy(s => s.SortOrder))
        {
            var points = await ResolveSeriesPointsAsync(s.HabitNodeId, s.MetricDefinitionId, from, to, granularity, ct)
                .ConfigureAwait(false);
            series.Add(new ChartSeriesModel(s.Label ?? "", s.Color, points));
        }

        var xAxis = new ChartAxisModel(config.XAxisLabel, IsTime: true, Format: "yyyy-MM-dd");
        var yAxis = new ChartAxisModel(config.YAxisLabel, IsTime: false, Format: null);
        return new ChartRenderModel(kind, chart.Name, series, xAxis, yAxis, theme);
    }

    private (DateOnly from, DateOnly to) ResolveRange(ChartConfig config)
    {
        var today = DateOnly.FromDateTime(clock.UtcNow.UtcDateTime);
        return config.Range switch
        {
            "30d" => (today.AddDays(-29), today),
            "1y" => (today.AddDays(-364), today),
            "custom" when TryDate(config.FromIso, out var f) && TryDate(config.ToIso, out var t) => (f, t),
            _ => (today.AddDays(-89), today), // "90d" and any fallback
        };

        static bool TryDate(string? iso, out DateOnly d)
            => DateOnly.TryParse(iso, CultureInfo.InvariantCulture, out d);
    }

    private async Task<IReadOnlyList<ChartPoint>> ResolveSeriesPointsAsync(
        Guid habitNodeId, Guid? metricId, DateOnly from, DateOnly to, TrendGranularity granularity, CancellationToken ct)
    {
        if (metricId is { } mid)
        {
            // Specific metric → daily sums; re-bucket weekly if needed.
            var daily = await completions.GetDailySumAsync(habitNodeId, mid, from, to, ct).ConfigureAwait(false);
            var buckets = daily.GroupBy(d => BucketKey(d.Date, granularity))
                .OrderBy(g => g.Key)
                .Select(g => new ChartPoint(DayNumber(g.Key), (double)g.Sum(x => x.Sum), g.Key.ToString("yyyy-MM-dd", CultureInfo.InvariantCulture)));
            return buckets.ToList();
        }

        var trend = await trends.GetTrendAsync(habitNodeId, from, to, granularity, ct).ConfigureAwait(false);
        return trend
            .Select(p => new ChartPoint(DayNumber(p.Bucket), (double)p.Value, p.Bucket.ToString("yyyy-MM-dd", CultureInfo.InvariantCulture)))
            .ToList();
    }

    private static DateOnly BucketKey(DateOnly date, TrendGranularity g) => g switch
    {
        TrendGranularity.Weekly => date.AddDays(-(((int)date.DayOfWeek + 6) % 7)),
        _ => date,
    };

    private static double DayNumber(DateOnly d) => d.ToDateTime(TimeOnly.MinValue).ToOADate();
}
```

> `ToOADate()` gives a stable numeric X for cartesian plotting while `Label` carries the human date. The renderer uses `Label` for axis ticks (Task 10).

using LiveChartsCore;
using LiveChartsCore.Defaults;
using LiveChartsCore.SkiaSharpView;
using LiveChartsCore.SkiaSharpView.Painting;
using Recordum.Plugins.Contracts.Charts;
using Recordum.Plugins.Contracts.ExtensionPoints;
using SkiaSharp;

namespace Recordum.Plugins.ChartEngine;

public sealed class ChartEngineRenderer : IChartRenderer
{
    public ExtensionPointId PointId => WellKnownExtensionPoints.ChartRenderer;
    public string PluginId => "io.recordum.chart-engine";

    public bool CanRender(ChartKind kind) => Enum.IsDefined(kind);

    /// <summary>Pure mapping ChartRenderModel → LiveCharts2 series. Testable without a control.</summary>
    public ISeries[] MapSeries(ChartRenderModel model)
    {
        ArgumentNullException.ThrowIfNull(model);
        if (model.Kind == ChartKind.Pie)
        {
            return model.Series
                .Select(s => (ISeries)new PieSeries<double>
                {
                    Name = s.Label,
                    Values = [s.Points.Sum(p => p.Y ?? 0d)],
                    Fill = Paint(s.ColorHex),
                })
                .ToArray();
        }

        return model.Series.Select(s => BuildCartesian(model.Kind, s)).ToArray();
    }

    private static ISeries BuildCartesian(ChartKind kind, ChartSeriesModel s)
    {
        var values = s.Points.Select(p => new ObservablePoint(p.X, p.Y)).ToArray();
        var stroke = Paint(s.ColorHex);
        return kind switch
        {
            ChartKind.Bar => new ColumnSeries<ObservablePoint> { Name = s.Label, Values = values, Fill = stroke },
            ChartKind.StackedBar => new StackedColumnSeries<ObservablePoint> { Name = s.Label, Values = values, Fill = stroke },
            ChartKind.Scatter => new ScatterSeries<ObservablePoint> { Name = s.Label, Values = values, Fill = stroke },
            ChartKind.Area => new LineSeries<ObservablePoint> { Name = s.Label, Values = values, Stroke = stroke, Fill = Paint(s.ColorHex, alpha: 64), GeometryFill = null, GeometryStroke = null },
            _ => new LineSeries<ObservablePoint> { Name = s.Label, Values = values, Stroke = stroke, Fill = null, GeometryFill = null, GeometryStroke = null },
        };
    }

    private static SolidColorPaint? Paint(string? hex, byte alpha = 255)
    {
        if (string.IsNullOrWhiteSpace(hex) || !SKColor.TryParse(hex, out var c)) return null;
        return new SolidColorPaint(c.WithAlpha(alpha));
    }

    public object CreateView(ChartRenderModel model) => throw new NotImplementedException("Task 10");
}

using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Models;
using Recordum.Plugins.Contracts.Charts;

namespace Recordum.WPF.ViewModels.Charts;

public sealed partial class ChartBuilderViewModel(IChartService charts) : ObservableObject
{
    private Guid? _editingId;

    [ObservableProperty] private string _name = "";
    [ObservableProperty] private ChartKind _selectedKind = ChartKind.Line;
    [ObservableProperty] private string _range = "90d";
    [ObservableProperty] private string _bucket = "daily";

    public ObservableCollection<ChartSeriesRowViewModel> Series { get; } = [];
    public IReadOnlyList<ChartKind> AvailableKinds { get; } = Enum.GetValues<ChartKind>();

    /// <summary>Raised after a successful save, carrying the persisted chart id.</summary>
    public event EventHandler<Guid>? Saved;

    public void LoadNew()
    {
        _editingId = null;
        Name = "";
        SelectedKind = ChartKind.Line;
        Range = "90d";
        Bucket = "daily";
        Series.Clear();
    }

    public async ValueTask LoadAsync(Guid chartId, CancellationToken ct = default)
    {
        var chart = await charts.GetAsync(chartId, ct).ConfigureAwait(true);
        if (chart is null) { LoadNew(); return; }
        _editingId = chart.Id;
        Name = chart.Name;
        SelectedKind = ChartKindParser.Parse(chart.ChartType);
        var cfg = ChartConfig.Parse(chart.ConfigJson);
        Range = cfg.Range;
        Bucket = cfg.Bucket;
        Series.Clear();
        foreach (var s in chart.Series.OrderBy(s => s.SortOrder))
            Series.Add(new ChartSeriesRowViewModel { HabitNodeId = s.HabitNodeId, MetricDefinitionId = s.MetricDefinitionId, Label = s.Label, Color = s.Color });
    }

    [RelayCommand]
    private void AddSeries() => Series.Add(new ChartSeriesRowViewModel());

    [RelayCommand]
    private void RemoveSeries(ChartSeriesRowViewModel row) => Series.Remove(row);

    [RelayCommand]
    private async Task SaveAsync(CancellationToken ct)
    {
        var configJson = new ChartConfig(Range: Range, Bucket: Bucket).ToJson();
        var rows = Series.Select((r, i) => new ChartSeries
        {
            Id = Guid.NewGuid(),
            HabitNodeId = r.HabitNodeId,
            MetricDefinitionId = r.MetricDefinitionId,
            Label = r.Label,
            Color = r.Color,
            SortOrder = i,
        }).ToList();

        Guid id;
        if (_editingId is { } existing)
        {
            await charts.UpdateAsync(existing, c =>
            {
                c.Name = Name;
                c.ChartType = ChartKindParser.ToToken(SelectedKind);
                c.ConfigJson = configJson;
            }, ct).ConfigureAwait(true);
            await charts.ReplaceSeriesAsync(existing, rows, ct).ConfigureAwait(true);
            id = existing;
        }
        else
        {
            id = await charts.CreateAsync(new Chart
            {
                Name = Name,
                ChartType = ChartKindParser.ToToken(SelectedKind),
                ConfigJson = configJson,
            }, rows, ct).ConfigureAwait(true);
            _editingId = id;
        }
        Saved?.Invoke(this, id);
    }
}

using System.Collections.ObjectModel;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using Recordum.Core.Abstractions.Services;
using Recordum.Data.Services.Queries;
using Recordum.Plugins.Contracts.Charts;
using Recordum.Plugins.Contracts.ExtensionPoints;
using Recordum.Plugins.Contracts.Hosting;

namespace Recordum.WPF.ViewModels.Charts;

public sealed partial class ChartsWorkspaceViewModel : ObservableObject
{
    private readonly IChartService _charts;
    private readonly IChartQueryService _query;
    private readonly IExtensionPointRegistry _registry;

    public ObservableCollection<ChartListItemViewModel> Charts { get; } = [];
    public ChartBuilderViewModel Builder { get; }

    [ObservableProperty] private ChartListItemViewModel? _selected;
    [ObservableProperty] private object? _renderedView;
    [ObservableProperty] private bool _isBuilderOpen;

    /// <summary>Theme flag pushed in by the host (MainWindow) so the renderer themes correctly.</summary>
    public bool IsDarkTheme { get; set; }

    public ChartsWorkspaceViewModel(
        IChartService charts, IChartQueryService query, IExtensionPointRegistry registry, ChartBuilderViewModel builder)
    {
        ArgumentNullException.ThrowIfNull(charts);
        ArgumentNullException.ThrowIfNull(query);
        ArgumentNullException.ThrowIfNull(registry);
        ArgumentNullException.ThrowIfNull(builder);
        _charts = charts;
        _query = query;
        _registry = registry;
        Builder = builder;
        Builder.Saved += async (_, id) => await OnBuilderSavedAsync(id).ConfigureAwait(true);
    }

    public async ValueTask LoadAsync(CancellationToken ct = default)
    {
        var all = await _charts.ListAsync(ct).ConfigureAwait(true);
        Charts.Clear();
        foreach (var c in all) Charts.Add(new ChartListItemViewModel(c));
    }

    public async ValueTask SelectAsync(ChartListItemViewModel? item, CancellationToken ct = default)
    {
        Selected = item;
        IsBuilderOpen = false;
        if (item is null) { RenderedView = null; return; }
        await RenderAsync(item.ChartId, ct).ConfigureAwait(true);
    }

    private async ValueTask RenderAsync(Guid chartId, CancellationToken ct)
    {
        var model = await _query.BuildModelAsync(chartId, IsDarkTheme, ct).ConfigureAwait(true);
        var renderer = _registry.GetExtensions(WellKnownExtensionPoints.ChartRenderer)
            .OfType<IChartRenderer>()
            .FirstOrDefault(r => r.CanRender(model.Kind));
        RenderedView = renderer?.CreateView(model);
    }

    [RelayCommand]
    private void New()
    {
        Builder.LoadNew();
        IsBuilderOpen = true;
    }

    [RelayCommand]
    private async Task Edit(ChartListItemViewModel item)
    {
        if (item is null) return;
        await Builder.LoadAsync(item.ChartId).ConfigureAwait(true);
        IsBuilderOpen = true;
    }

    [RelayCommand]
    private async Task Delete(ChartListItemViewModel item, CancellationToken ct)
    {
        if (item is null) return;
        await _charts.DeleteAsync(item.ChartId, ct).ConfigureAwait(true);
        Charts.Remove(item);
        if (Selected == item) { Selected = null; RenderedView = null; }
    }

    private async ValueTask OnBuilderSavedAsync(Guid id, CancellationToken ct = default)
    {
        IsBuilderOpen = false;
        await LoadAsync(ct).ConfigureAwait(true);
        var item = Charts.FirstOrDefault(c => c.ChartId == id);
        await SelectAsync(item, ct).ConfigureAwait(true);
    }
}
