# Milestone 3.10b — Backup / Restore Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship encrypted, portable `.recbak` backups (manual + scheduled auto) + restore (with pre-restore safety snapshot) + backup history + auto-backup settings, closing the offline-first promise. Spec: `docs/superpowers/archive/specs/2026-05-28-milestone-3.10b-backup-restore-design.md`.

**Architecture:** Core abstractions in `Recordum.Core/Backup/`. Concrete services in `Recordum.Data/Backup/` (talk to the DB, hence Data — same rationale as `IChartQueryService`/`IExportQueryService`). `.recbak` = `RCBK` header (KDF params + salt + nonce) + AES-256-GCM ciphertext of an inner ZIP (`manifest.json` + `db/recordum.db` plain dump + `attachments/`). Schema delta v2.6 → v2.7 (single additive migration `BackupLogV27`: one table + five `AppSetting` rows). WPF top-nav gains a Backup menu; a `BackupSchedulerHost` singleton ticks every 60 min for auto-backup.

**Tech Stack:** Konscious.Security.Cryptography.Argon2 (already in `Recordum.Data.csproj`), `System.IO.Compression`, `System.Security.Cryptography.AesGcm`, `System.Security.Cryptography.ProtectedData` (DPAPI), `SQLitePCLRaw.bundle_e_sqlcipher` (`sqlcipher_export`), CommunityToolkit.Mvvm, xUnit + FluentAssertions.

```csharp
using System.Collections.Concurrent;
using Recordum.Plugins.Contracts.Hosting;
using Recordum.Plugins.Contracts.Manifest;
using Recordum.Plugins.Contracts.Permissions;

namespace Recordum.Plugins.Management;

public sealed class PluginManager(
    IPluginHost host,
    IPermissionBroker broker,
    IExtensionPointRegistry registry) : IPluginManager
{
    private readonly ConcurrentDictionary<string, PluginManifest> _activated = new();

    public async Task EnableAsync(PluginManifest manifest, CancellationToken ct = default)
    {
        ArgumentNullException.ThrowIfNull(manifest);
        var decision = broker.Check(new PermissionRequest(manifest.Id, WellKnownPermissions.StorageRead));
        if (decision.Outcome != PermissionOutcome.Allowed)
            throw new InvalidOperationException($"Plugin '{manifest.Id}' denied by broker: {decision.Reason ?? "(no reason)"}");
        await host.LoadAsync(manifest, ct).ConfigureAwait(false);
        _activated[manifest.Id] = manifest;
    }

    public async Task DisableAsync(string pluginId, CancellationToken ct = default)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(pluginId);
        registry.UnregisterAll(pluginId);
        await host.UnloadAsync(pluginId, ct).ConfigureAwait(false);
        _activated.TryRemove(pluginId, out _);
    }

    public IReadOnlyCollection<string> ListEnabled() => _activated.Keys.ToArray();

    public IReadOnlyList<PluginManifest> ListActivatedManifests() => _activated.Values.ToArray();
}
```

```csharp
using System.Buffers.Binary;
using System.Security.Cryptography;
using System.Text;
using Konscious.Security.Cryptography;
using Recordum.Core.Abstractions.Security;
using Recordum.Core.Backup;

namespace Recordum.Data.Backup;

public sealed class AesGcmBackupCipher : IBackupCipher
{
    private static readonly byte[] Magic = "RCBK"u8.ToArray();
    private const byte FormatVersion = 0x01;
    private const byte KdfArgon2id = 0x01;
    private const int SaltLen = 16;
    private const int NonceLen = 12;
    private const int TagLen = 16;
    private const int KeyLen = 32;

    public async Task EncryptAsync(Stream input, Stream output, string passphrase, CancellationToken ct = default)
    {
        ArgumentNullException.ThrowIfNull(input);
        ArgumentNullException.ThrowIfNull(output);
        ArgumentException.ThrowIfNullOrEmpty(passphrase);

        ushort memMib = 64;
        byte iters = 4;
        byte parallel = 2;
        var salt = RandomNumberGenerator.GetBytes(SaltLen);
        var nonce = RandomNumberGenerator.GetBytes(NonceLen);

        var header = new byte[4 + 1 + 3 + 1 + 2 + 1 + 1 + SaltLen + NonceLen];
        var span = header.AsSpan();
        Magic.CopyTo(span); span = span[4..];
        span[0] = FormatVersion; span = span[1..];
        span[..3].Clear(); span = span[3..];      // reserved
        span[0] = KdfArgon2id; span = span[1..];
        BinaryPrimitives.WriteUInt16BigEndian(span, memMib); span = span[2..];
        span[0] = iters; span = span[1..];
        span[0] = parallel; span = span[1..];
        salt.CopyTo(span); span = span[SaltLen..];
        nonce.CopyTo(span);

        await output.WriteAsync(header, ct).ConfigureAwait(false);

        var key = await DeriveKeyAsync(passphrase, salt, memMib, iters, parallel).ConfigureAwait(false);

        // Materialize plaintext (backups are bounded; streaming GCM here would add ad-hoc framing).
        using var ms = new MemoryStream();
        await input.CopyToAsync(ms, ct).ConfigureAwait(false);
        var plain = ms.ToArray();
        var ct_buf = new byte[plain.Length];
        var tag = new byte[TagLen];
        using (var gcm = new AesGcm(key, TagLen))
        {
            gcm.Encrypt(nonce, plain, ct_buf, tag);
        }
        await output.WriteAsync(ct_buf, ct).ConfigureAwait(false);
        await output.WriteAsync(tag, ct).ConfigureAwait(false);
        CryptographicOperations.ZeroMemory(key);
    }

    public async Task DecryptAsync(Stream input, Stream output, string passphrase, CancellationToken ct = default)
    {
        ArgumentNullException.ThrowIfNull(input);
        ArgumentNullException.ThrowIfNull(output);
        ArgumentException.ThrowIfNullOrEmpty(passphrase);

        var header = new byte[4 + 1 + 3 + 1 + 2 + 1 + 1 + SaltLen + NonceLen];
        var read = await input.ReadAtLeastAsync(header, header.Length, throwOnEndOfStream: false, ct).ConfigureAwait(false);
        if (read < header.Length || !header.AsSpan(0, 4).SequenceEqual(Magic))
            throw new RestoreException("invalid_format", "Not a Recordum backup file.");
        if (header[4] != FormatVersion)
            throw new RestoreException("invalid_format", $"Unsupported backup version {header[4]}.");
        if (header[8] != KdfArgon2id)
            throw new RestoreException("invalid_format", $"Unsupported KDF {header[8]}.");

        var memMib = BinaryPrimitives.ReadUInt16BigEndian(header.AsSpan(9, 2));
        var iters = header[11];
        var parallel = header[12];
        var salt = header.AsSpan(13, SaltLen).ToArray();
        var nonce = header.AsSpan(13 + SaltLen, NonceLen).ToArray();

        using var ms = new MemoryStream();
        await input.CopyToAsync(ms, ct).ConfigureAwait(false);
        var rest = ms.ToArray();
        if (rest.Length < TagLen)
            throw new RestoreException("invalid_format", "Truncated backup.");

        var cipherLen = rest.Length - TagLen;
        var cipherText = rest.AsSpan(0, cipherLen).ToArray();
        var tag = rest.AsSpan(cipherLen, TagLen).ToArray();

        var key = await DeriveKeyAsync(passphrase, salt, memMib, iters, parallel).ConfigureAwait(false);
        var plain = new byte[cipherLen];
        try
        {
            using var gcm = new AesGcm(key, TagLen);
            gcm.Decrypt(nonce, cipherText, tag, plain);
        }
        catch (AuthenticationTagMismatchException ex)
        {
            throw new RestoreException("invalid_passphrase", "Wrong passphrase or backup integrity check failed.")
                .Pipe(_ => { _ = ex; });
        }
        finally
        {
            CryptographicOperations.ZeroMemory(key);
        }

        await output.WriteAsync(plain, ct).ConfigureAwait(false);
    }

    private static async Task<byte[]> DeriveKeyAsync(string passphrase, byte[] salt, ushort memMib, byte iters, byte parallel)
    {
        using var argon = new Argon2id(Encoding.UTF8.GetBytes(passphrase))
        {
            Salt = salt,
            DegreeOfParallelism = parallel,
            Iterations = iters,
            MemorySize = memMib * 1024,
        };
        return await argon.GetBytesAsync(KeyLen).ConfigureAwait(false);
    }
}

}
```

```csharp
using System.IO.Compression;
using System.Text.Json;
using Recordum.Core.Backup;

namespace Recordum.Data.Backup;

public sealed class BackupArchiveWriter
{
    private static readonly JsonSerializerOptions Json = new()
    {
        PropertyNamingPolicy = JsonNamingPolicy.SnakeCaseLower,
        WriteIndented = true,
    };

    public async Task WriteAsync(Stream destination, BackupManifest manifest, string dbFilePath, string attachmentsDir, CancellationToken ct = default)
    {
        using var zip = new ZipArchive(destination, ZipArchiveMode.Create, leaveOpen: true);

        var manifestEntry = zip.CreateEntry("manifest.json", CompressionLevel.Optimal);
        await using (var es = manifestEntry.Open())
        await using (var sw = new StreamWriter(es))
        {
            await sw.WriteAsync(JsonSerializer.Serialize(manifest, Json));
        }

        if (File.Exists(dbFilePath))
        {
            var dbEntry = zip.CreateEntry("db/recordum.db", CompressionLevel.Optimal);
            await using var es = dbEntry.Open();
            await using var fs = File.OpenRead(dbFilePath);
            await fs.CopyToAsync(es, ct).ConfigureAwait(false);
        }

        if (Directory.Exists(attachmentsDir))
        {
            foreach (var path in Directory.EnumerateFiles(attachmentsDir, "*", SearchOption.TopDirectoryOnly))
            {
                var entry = zip.CreateEntry($"attachments/{Path.GetFileName(path)}", CompressionLevel.Optimal);
                await using var es = entry.Open();
                await using var fs = File.OpenRead(path);
                await fs.CopyToAsync(es, ct).ConfigureAwait(false);
            }
        }
    }
}
```

```csharp
using System.IO.Compression;
using System.Text.Json;
using Recordum.Core.Backup;

namespace Recordum.Data.Backup;

public sealed class BackupArchiveReader
{
    private static readonly JsonSerializerOptions Json = new()
    {
        PropertyNamingPolicy = JsonNamingPolicy.SnakeCaseLower,
    };

    public async Task<BackupManifest> ReadManifestAsync(string zipPath, CancellationToken ct = default)
    {
        using var zip = ZipFile.OpenRead(zipPath);
        var entry = zip.GetEntry("manifest.json")
            ?? throw new RestoreException("invalid_format", "manifest.json missing");
        await using var s = entry.Open();
        using var sr = new StreamReader(s);
        var json = await sr.ReadToEndAsync(ct).ConfigureAwait(false);
        var manifest = JsonSerializer.Deserialize<BackupManifest>(json, Json)
            ?? throw new RestoreException("invalid_format", "manifest.json malformed");
        return manifest;
    }

    public async Task ExtractDbAsync(string zipPath, string destinationFile, CancellationToken ct = default)
    {
        using var zip = ZipFile.OpenRead(zipPath);
        var entry = zip.GetEntry("db/recordum.db")
            ?? throw new RestoreException("invalid_format", "db/recordum.db missing");
        await using var src = entry.Open();
        await using var dst = File.Create(destinationFile);
        await src.CopyToAsync(dst, ct).ConfigureAwait(false);
    }

    public async Task ExtractAttachmentsAsync(string zipPath, string destinationDir, CancellationToken ct = default)
    {
        Directory.CreateDirectory(destinationDir);
        using var zip = ZipFile.OpenRead(zipPath);
        foreach (var entry in zip.Entries)
        {
            if (!entry.FullName.StartsWith("attachments/", StringComparison.Ordinal)) continue;
            if (entry.FullName.EndsWith('/')) continue;
            var name = Path.GetFileName(entry.FullName);
            var dst = Path.Combine(destinationDir, name);
            await using var src = entry.Open();
            await using var fs = File.Create(dst);
            await src.CopyToAsync(fs, ct).ConfigureAwait(false);
        }
    }
}
```

```csharp
using Microsoft.Data.Sqlite;
using Microsoft.EntityFrameworkCore;
using Recordum.Core.Abstractions;
using Recordum.Core.Abstractions.Security;
using Recordum.Core.Abstractions.Services;
using Recordum.Core.Backup;
using Recordum.Core.Models;
using Recordum.Plugins.Contracts.Hosting;

namespace Recordum.Data.Backup;

public sealed class BackupService(
    IDbContextFactory<RecordumDbContext> dbFactory,
    IDbKeyProvider keyProvider,
    IDbStoragePaths paths,
    IBackupCipher cipher,
    BackupArchiveWriter writer,
    IBackupLogService log,
    IPluginManager plugins,
    IClock clock,
    string appVersion) : IBackupService
{
    private const string CurrentSchemaVersion = "2.7";

    public async Task<BackupResult> CreateAsync(string destinationPath, string passphrase, BackupOptions options, CancellationToken ct = default)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(destinationPath);
        ArgumentException.ThrowIfNullOrWhiteSpace(passphrase);

        var parent = Path.GetDirectoryName(destinationPath);
        if (!string.IsNullOrEmpty(parent)) Directory.CreateDirectory(parent);

        var tempDir = Path.Combine(Path.GetTempPath(), $"recordum-bk-{Guid.NewGuid():N}");
        Directory.CreateDirectory(tempDir);
        var tempDb = Path.Combine(tempDir, "recordum.db");
        var tempAttachments = Path.Combine(tempDir, "attachments");
        Directory.CreateDirectory(tempAttachments);
        var tempZip = Path.Combine(tempDir, "inner.zip");

        try
        {
            // 1. Plain DB dump via VACUUM INTO. Retry on busy.
            await DumpDbAsync(tempDb, ct).ConfigureAwait(false);

            // 2. Copy attachments.
            if (Directory.Exists(((Recordum.WPF.Bootstrap.RecordumOptions)paths).AttachmentsDirectory))
            {
                foreach (var f in Directory.EnumerateFiles(((Recordum.WPF.Bootstrap.RecordumOptions)paths).AttachmentsDirectory))
                    File.Copy(f, Path.Combine(tempAttachments, Path.GetFileName(f)), overwrite: true);
            }

            // 3. Build manifest.
            var manifest = await BuildManifestAsync(options.Kind, ct).ConfigureAwait(false);

            // 4. Stream-zip.
            await using (var zfs = File.Create(tempZip))
                await writer.WriteAsync(zfs, manifest, tempDb, tempAttachments, ct).ConfigureAwait(false);

            // 5. Encrypt envelope to destination.
            await using (var src = File.OpenRead(tempZip))
            await using (var dst = File.Create(destinationPath))
                await cipher.EncryptAsync(src, dst, passphrase, ct).ConfigureAwait(false);

            // 6. Log row.
            var id = Guid.NewGuid();
            var row = new BackupLog
            {
                Id = id,
                CreatedAt = clock.UtcNow,
                Kind = options.Kind.ToString(),
                Path = destinationPath,
                SizeBytes = new FileInfo(destinationPath).Length,
                SchemaVersion = manifest.SchemaVersion,
                AppVersion = manifest.AppVersion,
            };
            await log.RecordAsync(row, ct).ConfigureAwait(false);

            // 7. Retention for Auto.
            if (options.Kind == BackupKind.Auto)
                await ApplyRetentionAsync(ct).ConfigureAwait(false);

            return new BackupResult(id, destinationPath, row.SizeBytes, manifest);
        }
        catch (Exception ex) when (ex is not BackupException)
        {
            // Best-effort cleanup of half-written destination.
            try { if (File.Exists(destinationPath)) File.Delete(destinationPath); } catch { }
            throw;
        }
        finally
        {
            ShredDirectory(tempDir);
        }
    }

    private async Task DumpDbAsync(string targetPath, CancellationToken ct)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct).ConfigureAwait(false);
        var conn = (SqliteConnection)db.Database.GetDbConnection();
        if (conn.State != System.Data.ConnectionState.Open) await conn.OpenAsync(ct).ConfigureAwait(false);

        for (var attempt = 0; ; attempt++)
        {
            try
            {
                using var cmd = conn.CreateCommand();
                cmd.CommandText = "VACUUM INTO $p";
                cmd.Parameters.AddWithValue("$p", targetPath);
                await cmd.ExecuteNonQueryAsync(ct).ConfigureAwait(false);
                return;
            }
            catch (SqliteException ex) when (ex.SqliteErrorCode == 5 /* SQLITE_BUSY */ && attempt < 2)
            {
                await Task.Delay(100, ct).ConfigureAwait(false);
            }
            catch (SqliteException ex) when (ex.SqliteErrorCode == 5)
            {
                throw new BackupException("db_busy_retry_later", ex.Message);
            }
        }
    }

    private async Task<BackupManifest> BuildManifestAsync(BackupKind kind, CancellationToken ct)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct).ConfigureAwait(false);
        var habits = await db.HabitNodes.IgnoreQueryFilters().CountAsync(x => x.NodeType == Core.Enums.HabitNodeType.Habit && x.DeletedAt == null, ct).ConfigureAwait(false);
        var completions = await db.Completions.CountAsync(ct).ConfigureAwait(false);
        var notes = await db.Notes.CountAsync(ct).ConfigureAwait(false);
        var attachments = await db.Attachments.CountAsync(ct).ConfigureAwait(false);

        var plugins_ = plugins.ListActivatedManifests()
            .Select(m => new PluginManifestSnapshot(m.Id, m.Version, true, false))
            .ToArray();

        return new BackupManifest(
            SchemaVersion: CurrentSchemaVersion,
            AppVersion: appVersion,
            CreatedAtUtc: clock.UtcNow,
            Kind: kind,
            SizeBytesUncompressed: 0,
            Counts: new BackupCounts(habits, completions, notes, attachments),
            Plugins: plugins_,
            Permissions: new PermissionState(new Dictionary<string, IReadOnlyList<string>>()));
    }

    private async Task ApplyRetentionAsync(CancellationToken ct)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct).ConfigureAwait(false);
        var retention = await db.AppSettings.AsNoTracking()
            .Where(x => x.Key == "backup.auto.retention_count")
            .Select(x => x.Value).FirstOrDefaultAsync(ct).ConfigureAwait(false);
        if (!int.TryParse(retention, out var count) || count <= 0) return;

        var autos = await db.BackupLogs.AsNoTracking()
            .Where(x => x.Kind == "Auto")
            .OrderByDescending(x => x.CreatedAt)
            .ToListAsync(ct).ConfigureAwait(false);

        foreach (var stale in autos.Skip(count))
        {
            try { if (File.Exists(stale.Path)) File.Delete(stale.Path); } catch { }
            await log.DeleteAsync(stale.Id, ct).ConfigureAwait(false);
        }
    }

    private static void ShredDirectory(string dir)
    {
        if (!Directory.Exists(dir)) return;
        try
        {
            foreach (var f in Directory.EnumerateFiles(dir, "*", SearchOption.AllDirectories))
            {
                try
                {
                    var fi = new FileInfo(f);
                    using var fs = File.Open(f, FileMode.Open, FileAccess.Write);
                    var zero = new byte[Math.Min(fi.Length, 65_536)];
                    var remaining = fi.Length;
                    while (remaining > 0)
                    {
                        var n = (int)Math.Min(zero.Length, remaining);
                        fs.Write(zero, 0, n);
                        remaining -= n;
                    }
                }
                catch { }
            }
            Directory.Delete(dir, recursive: true);
        }
        catch { }
    }
}
```

```csharp
        ArgumentException.ThrowIfNullOrWhiteSpace(sourcePath);
        ArgumentException.ThrowIfNullOrWhiteSpace(passphrase);
        if (!File.Exists(sourcePath))
            throw new RestoreException("invalid_format", $"Backup file not found: {sourcePath}");

                .Where(p => !activated.Contains(p.PluginId))
                .Select(p => $"Plugin '{p.PluginId}' not currently activated.")
                .ToArray();

            return new RestorePreview(manifest, warnings);
        }
        finally
        {
            try { if (File.Exists(tempZip)) File.Delete(tempZip); } catch { }
        }
```

```csharp
public async Task<RestoreResult> RestoreAsync(string sourcePath, string passphrase, RestoreOptions options, CancellationToken ct = default)
{
    var preview = await PreviewAsync(sourcePath, passphrase, ct).ConfigureAwait(false);
    var attachmentsDir = ((Recordum.WPF.Bootstrap.RecordumOptions)paths).AttachmentsDirectory;
    var backupsDir = ((Recordum.WPF.Bootstrap.RecordumOptions)paths).BackupsDirectory;

    // 1. Safety snapshot.
    if (!options.SkipSafetySnapshot)
    {
        var snapshotPath = Path.Combine(backupsDir, $"pre-restore-{clock.UtcNow:yyyyMMdd-HHmmss}.recbak");
        try
        {
            await backupService.CreateAsync(snapshotPath, passphrase, new BackupOptions(BackupKind.PreRestore), ct).ConfigureAwait(false);
        }
        catch (Exception ex)
        {
            throw new RestoreException("safety_snapshot_failed", ex.Message);
        }
    }

    // 2. Drain pool so we can replace the DB file.
    Microsoft.Data.Sqlite.SqliteConnection.ClearAllPools();

    // 3. Decrypt to temp zip and extract.
    var tempZip = Path.Combine(Path.GetTempPath(), $"recordum-rs-{Guid.NewGuid():N}.zip");
    var tempPlainDb = Path.Combine(Path.GetTempPath(), $"recordum-rs-{Guid.NewGuid():N}.db");
    var dbPath = ((Recordum.WPF.Bootstrap.RecordumOptions)paths).DatabasePath;
    try
    {
        await using (var src = File.OpenRead(sourcePath))
        await using (var dst = File.Create(tempZip))
            await cipher.DecryptAsync(src, dst, passphrase, ct).ConfigureAwait(false);

        await reader.ExtractDbAsync(tempZip, tempPlainDb, ct).ConfigureAwait(false);

        // Clear current attachments + DB.
        if (File.Exists(dbPath)) File.Delete(dbPath);
        if (Directory.Exists(attachmentsDir))
            foreach (var f in Directory.EnumerateFiles(attachmentsDir)) File.Delete(f);
        await reader.ExtractAttachmentsAsync(tempZip, attachmentsDir, ct).ConfigureAwait(false);

        // 4. Re-encrypt DB into destinationPath with the current cipher key.
        using var key = await keyProvider.GetKeyAsync(ct).ConfigureAwait(false);
        await ReEncryptDbAsync(tempPlainDb, dbPath, key.Bytes).ConfigureAwait(false);
    }
    finally
    {
        try { if (File.Exists(tempZip)) File.Delete(tempZip); } catch { }
        try { if (File.Exists(tempPlainDb)) File.Delete(tempPlainDb); } catch { }
    }

    // 5. Plugin reconciliation = warnings already in preview; surface as UnavailablePlugins.
    var activated = plugins.ListActivatedManifests().Select(m => m.Id).ToHashSet(StringComparer.Ordinal);
    var unavailable = preview.Manifest.Plugins
        .Where(p => !activated.Contains(p.PluginId))
        .Select(p => p.PluginId)
        .ToArray();
    return new RestoreResult(unavailable);
}

private static async Task ReEncryptDbAsync(string plainSourcePath, string destinationPath, byte[] keyBytes)
{
    // ATTACH the destination with the cipher key, then sqlcipher_export.
    var hexKey = Convert.ToHexString(keyBytes);
    await using var conn = new Microsoft.Data.Sqlite.SqliteConnection($"Data Source={plainSourcePath}");
    await conn.OpenAsync();
    using (var attach = conn.CreateCommand())
    {
        attach.CommandText = $"ATTACH DATABASE @p AS enc KEY \"x'{hexKey}'\"";
        attach.Parameters.AddWithValue("@p", destinationPath);
        await attach.ExecuteNonQueryAsync();
    }
    using (var export = conn.CreateCommand())
    {
        export.CommandText = "SELECT sqlcipher_export('enc')";
        await export.ExecuteScalarAsync();
    }
    using (var detach = conn.CreateCommand())
    {
        detach.CommandText = "DETACH DATABASE enc";
        await detach.ExecuteNonQueryAsync();
    }
}
```

```csharp
using System.Globalization;
using System.Security.Cryptography;
using System.Text;
using Microsoft.EntityFrameworkCore;
using Recordum.Core.Abstractions;
using Recordum.Core.Backup;

namespace Recordum.Data.Backup;

public sealed class BackupScheduler(
    IDbContextFactory<RecordumDbContext> dbFactory,
    IBackupService backupService,
    Recordum.WPF.Bootstrap.RecordumOptions options,
    IClock clock) : IBackupScheduler
{
    public bool IsEnabled { get; private set; } = true;
    public DateTime? NextRunUtc { get; private set; }

    public async Task RunDueAsync(CancellationToken ct = default)
    {
        await using var db = await dbFactory.CreateDbContextAsync(ct).ConfigureAwait(false);
        var settings = await db.AppSettings.AsTracking()
            .Where(x => x.Key.StartsWith("backup.auto."))
            .ToDictionaryAsync(x => x.Key, ct).ConfigureAwait(false);

        var enabled = settings["backup.auto.enabled"].Value == "true";
        IsEnabled = enabled;
        if (!enabled) return;

        if (!File.Exists(options.AutoBackupKeyPath))
        {
            // inert state — quiet skip
            return;
        }

        string passphrase;
        try
        {
#pragma warning disable CA1416
            var wrapped = await File.ReadAllBytesAsync(options.AutoBackupKeyPath, ct).ConfigureAwait(false);
            var raw = ProtectedData.Unprotect(wrapped, optionalEntropy: null, DataProtectionScope.CurrentUser);
            passphrase = Encoding.UTF8.GetString(raw);
            CryptographicOperations.ZeroMemory(raw);
#pragma warning restore CA1416
        }
        catch
        {
            settings["backup.auto.enabled"].Value = "false";
            settings["backup.auto.enabled"].UpdatedAt = clock.UtcNow;
            await db.SaveChangesAsync(ct).ConfigureAwait(false);
            IsEnabled = false;
            return;
        }

        var frequencyDays = int.TryParse(settings["backup.auto.frequency_days"].Value, out var d) ? d : 7;
        var lastRunStr = settings["backup.auto.last_run_utc"].Value;
        var lastRun = string.IsNullOrWhiteSpace(lastRunStr) ? DateTime.MinValue
            : DateTime.Parse(lastRunStr, CultureInfo.InvariantCulture, DateTimeStyles.AssumeUniversal | DateTimeStyles.AdjustToUniversal);
        NextRunUtc = lastRun.AddDays(frequencyDays);

        if (clock.UtcNow < NextRunUtc) return;

        var folder = string.IsNullOrWhiteSpace(settings["backup.auto.path"].Value)
            ? options.BackupsDirectory
            : settings["backup.auto.path"].Value;
        Directory.CreateDirectory(folder);
        var dest = Path.Combine(folder, $"auto-{clock.UtcNow:yyyyMMdd-HHmmss}.recbak");

        await backupService.CreateAsync(dest, passphrase, new BackupOptions(BackupKind.Auto), ct).ConfigureAwait(false);

        settings["backup.auto.last_run_utc"].Value = clock.UtcNow.ToString("O");
        settings["backup.auto.last_run_utc"].UpdatedAt = clock.UtcNow;
        await db.SaveChangesAsync(ct).ConfigureAwait(false);
    }
}
```

```csharp
            UnavailablePlugins.Clear();
            foreach (var p in r.UnavailablePlugins) UnavailablePlugins.Add(p);
            IsCompleted = true;
        }
        catch (RestoreException ex) { ErrorMessage = ex.Code; }
        finally { IsRunning = false; }
```

`src/Recordum.WPF/Bootstrap/BackupSchedulerHost.cs`:

```csharp
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using Recordum.Core.Backup;

namespace Recordum.WPF.Bootstrap;

public sealed class BackupSchedulerHost(IServiceScopeFactory scopes, ILogger<BackupSchedulerHost> log) : IDisposable
{
    private PeriodicTimer? _timer;
    private CancellationTokenSource? _cts;
    private Task? _loop;

    public void Initialize()
    {
        CleanupOrphanTempDirs();
        _cts = new CancellationTokenSource();
        _timer = new PeriodicTimer(TimeSpan.FromMinutes(60));
        _loop = Task.Run(() => LoopAsync(_cts.Token));
    }

    private async Task LoopAsync(CancellationToken ct)
    {
        // Fire once at startup, then every tick.
        await TickAsync(ct).ConfigureAwait(false);
        while (await _timer!.WaitForNextTickAsync(ct).ConfigureAwait(false))
            await TickAsync(ct).ConfigureAwait(false);
    }

    private async Task TickAsync(CancellationToken ct)
    {
        try
        {
            await using var scope = scopes.CreateAsyncScope();
            var sched = scope.ServiceProvider.GetRequiredService<IBackupScheduler>();
            await sched.RunDueAsync(ct).ConfigureAwait(false);
        }
        catch (OperationCanceledException) { }
        catch (Exception ex)
        {
            log.LogWarning(ex, "auto-backup tick failed");
        }
    }

    private static void CleanupOrphanTempDirs()
    {
        try
        {
            var cutoff = DateTime.UtcNow.AddHours(-1);
            foreach (var d in Directory.EnumerateDirectories(Path.GetTempPath(), "recordum-bk-*"))
            {
                try
                {
                    if (Directory.GetCreationTimeUtc(d) < cutoff)
                        Directory.Delete(d, recursive: true);
                }
                catch { }
            }
        }
        catch { }
    }

    public void Dispose()
    {
        _cts?.Cancel();
        _timer?.Dispose();
        _cts?.Dispose();
    }
}
```

```csharp
using FluentAssertions;
using Microsoft.EntityFrameworkCore;

    [Fact]
    public async Task Full_pipeline_backup_then_restore_recovers_state()
    {
        var temp = Directory.CreateTempSubdirectory("rec-e2e-");
        try
        {
            // Build DI rooted at a temp folder
            var options = new RecordumOptions
            {
                DatabasePath = Path.Combine(temp.FullName, "recordum.db"),
                KeyFilePath = Path.Combine(temp.FullName, "db.key"),
                SaltFilePath = Path.Combine(temp.FullName, "db.salt"),
                AttachmentsDirectory = Path.Combine(temp.FullName, "attachments"),
                BackupsDirectory = Path.Combine(temp.FullName, "backups"),
                AutoBackupKeyPath = Path.Combine(temp.FullName, "autobackup.key"),
            };
            Directory.CreateDirectory(options.AttachmentsDirectory);
            Directory.CreateDirectory(options.BackupsDirectory);

            var services = new ServiceCollection().AddRecordumServices(options).BuildServiceProvider();
            await services.GetRequiredService<ApplicationInitializer>().InitializeAsync();

            // Snapshot pre-backup row counts
            int habitsBefore, settingsBefore;
            using (var scope1 = services.CreateScope())
            await using (var db = await scope1.ServiceProvider.GetRequiredService<IDbContextFactory<RecordumDbContext>>().CreateDbContextAsync())
            {
                habitsBefore = await db.HabitNodes.IgnoreQueryFilters().CountAsync();
                settingsBefore = await db.AppSettings.CountAsync();
            }

            // Create backup
            string dest;
            using (var scope2 = services.CreateScope())
            {
                var backup = scope2.ServiceProvider.GetRequiredService<IBackupService>();
                dest = Path.Combine(options.BackupsDirectory, "e2e.recbak");
                await backup.CreateAsync(dest, "passphrase-1234567890", new BackupOptions(BackupKind.Manual));
            }
            File.Exists(dest).Should().BeTrue();

            // Mutate DB (delete a row)
            using (var scope3 = services.CreateScope())
            await using (var db = await scope3.ServiceProvider.GetRequiredService<IDbContextFactory<RecordumDbContext>>().CreateDbContextAsync())
            {
                if (await db.HabitNodes.AnyAsync())
                    await db.HabitNodes.Take(1).ExecuteDeleteAsync();
            }

            // Restore
            using (var scope4 = services.CreateScope())
            {
                var restore = scope4.ServiceProvider.GetRequiredService<IRestoreService>();
                var result = await restore.RestoreAsync(dest, "passphrase-1234567890", new RestoreOptions(SkipSafetySnapshot: false));
                result.UnavailablePlugins.Should().BeEmpty();
            }

            // Verify counts restored
            using (var scope5 = services.CreateScope())
            await using (var db = await scope5.ServiceProvider.GetRequiredService<IDbContextFactory<RecordumDbContext>>().CreateDbContextAsync())
            {
                (await db.HabitNodes.IgnoreQueryFilters().CountAsync()).Should().Be(habitsBefore);
                (await db.AppSettings.CountAsync()).Should().Be(settingsBefore);
            }

            // Safety snapshot was created
            Directory.EnumerateFiles(options.BackupsDirectory, "pre-restore-*.recbak").Should().NotBeEmpty();
        }
        finally
        {
            try { Directory.Delete(temp.FullName, recursive: true); } catch { }
        }
    }
}
```
